# OptiSim architecture 

## 1. Class diagram


```mermaid
classDiagram
    class SimulationApp {
        -SimulationController controller
        -MainWindow main_window
        -list open_windows
        +start()
        +handle_simulation_finished(payload)
        +handle_error(err_msg)
        +open_strategy_window(track_name)
    }

    class MainWindow {
        -SimulationController controller
        -LoadingOverlay loading_overlay
        +openCompareRequested: pyqtSignal
        +openStrategyRequested: pyqtSignal
        +on_run_single_clicked()
        +on_lap_map_clicked()
        -_get_car_params() dict
    }

    %% Data Transfer Objects (DTOs)
    class BaseRequest {
        +dict car_params
        +float ds
    }
    class SingleLapRequest
    class EnduranceRequest
    class LapMapTestRequest
    class StrategyOptRequest
    class DeploymentOptRequest
    
    BaseRequest <|-- SingleLapRequest
    BaseRequest <|-- EnduranceRequest
    BaseRequest <|-- LapMapTestRequest
    BaseRequest <|-- StrategyOptRequest
    StrategyOptRequest <|-- DeploymentOptRequest

    class SimulationController {
        -SimulationWorker active_worker
        +simulationStarted: pyqtSignal
        +simulationProgress: pyqtSignal
        +simulationFinished: pyqtSignal
        +simulationError: pyqtSignal
        +handle_request(req: BaseRequest)
        -_create_solver(...)
    }

    class SimulationWorker {
        -SimulationTask task
        +progress: pyqtSignal
        +finished: pyqtSignal
        +error: pyqtSignal
        +run()
    }

    %% Tasks
    class SimulationTask {
        <<Abstract>>
        +execute(progress_callback) SolverPayload
    }
    class SingleLapTask
    class LaptimeMapTask
    class StrategyOptTask
    class DeploymentOptTask
    
    SimulationTask <|-- SingleLapTask
    SimulationTask <|-- LaptimeMapTask
    SimulationTask <|-- StrategyOptTask
    SimulationTask <|-- DeploymentOptTask

    class SolverPayload {
        +str kind
        +object data
        +dict meta
    }

    %% Relationships
    SimulationApp *-- SimulationController : Holds instance
    SimulationApp *-- MainWindow : Holds instance
    
    MainWindow ..> BaseRequest : Creates DTO
    MainWindow --> SimulationController : Calls handle_request(req)
    
    SimulationController ..> SimulationTask : Creates based on Request
    SimulationController *-- SimulationWorker : Manages thread
    
    SimulationWorker o-- SimulationTask : Executes
    SimulationTask ..> SolverPayload : Returns
```

## 2. State diagram 


```mermaid

stateDiagram-v2
    [*] --> Idle: Application waits for user

    state "UI Layer (MainWindow / StrategyWindow)" as UI {
        Idle --> ReadInput: Button click (e.g., Run Single Lap)
        ReadInput --> CreateRequest: Create DTO (e.g., SingleLapRequest)
        CreateRequest --> SendRequest: controller.handle_request(req)
    }

    state "Controller Layer (Assembly)" as Controller {
        SendRequest --> AssembleCar: Initialize Car() from req.car_params
        AssembleCar --> Routing: What type of Request is it?
        
        Routing --> CreateSingleTask: type(req) is SingleLapRequest
        Routing --> CreateLapMapTask: type(req) is LapMapTestRequest
        Routing --> CreateStrategyTask: type(req) is StrategyOptRequest
        Routing --> CreateDeploymentTask: type(req) is DeploymentOptRequest
        
        CreateSingleTask --> StartWorker
        CreateLapMapTask --> StartWorker
        CreateStrategyTask --> StartWorker
        CreateDeploymentTask --> StartWorker
        
        StartWorker --> active_worker.start()
    }

    state "Loading (MainWindow)" as Loading {
        AssembleCar --> ShowLoading: emit simulationStarted()
    }

    state "Worker Thread (SimulationWorker)" as Worker {
        active_worker.start() --> TaskExecute: task.execute(callback)
        TaskExecute --> TaskExecute: emit progress(current, total)
        TaskExecute --> CreatePayload: Computation finished
    }

    state "Results Processing (SimulationApp)" as Results {
        CreatePayload --> Finish: emit finished(SolverPayload)
        
        Finish --> CreateWindow: App catches signal and checks payload.kind
        CreateWindow --> ShowWindow: App stores and shows window (Results, Sensitivity, etc.)
    }

    ShowWindow --> Idle
    Finish --> HideLoading: UI catches signal and hides overlay
    HideLoading --> Idle
```

class SimulationApp:
    def __init__(self):
        
        # Create controller
        self.controller = SimulationController()
        
        # Create main window and pass the controller to it
        self.main_window = MainWindow(self.controller)
        
        # here we track references to windows
        self.open_windows = []

        # sginals from controller to update UI
        self.controller.simulationFinished.connect(self.handle_simulation_finished)
        self.controller.simulationError.connect(self.handle_error)
        
        # signals from main window
        self.main_window.openStrategyRequested.connect(self.open_strategy_window)
        self.main_window.openCompareRequested.connect(self.open_compare_window)

    def start(self):
        self.main_window.show()

    # ==========================================
    # opening different windows based on user actions
    # ==========================================
    def open_strategy_window(self, track_name):
        
        
        name = track_name + "_laptime_map"
        tracks_dir = "data/tracks/lapmaps/"
        filepath = os.path.join(tracks_dir, f"{name}.npz")

        if not os.path.exists(filepath):
            QMessageBox.warning(
                self.main_window, 
                "Missing data", 
                f"Laptime map '{name}.npz' not found.\nPlease generate it first using 'Generate laptime map' button."
            )
            return
        
        try:
            win = LaptimeAppMainWindow(filepath)
            
            # get actual car params from  MainWindow
            car_params = self.main_window._get_car_params()
            ds = self.main_window.input_ds.value()
            laps = self.main_window.car_params["laps"]
            
            
            # connect strategy optimization button
            win.optimization_requested.connect(
                lambda r, d, min_e, max_e, m_data: self.controller.handle_request(
                    StrategyOptRequest(
                        car_params=car_params,
                        ds=ds,
                        laps=laps,
                        regen_val=r,
                        delta_val=d,
                        min_e_target=min_e,
                        max_e_target=max_e,
                        map_data=m_data
                    )
                )
            )
            
            # connect deployment button
            win.deployment_requested.connect(
                lambda r, d, min_e, max_e, m_data: self.controller.handle_request(
                    DeploymentOptRequest(
                        car_params=car_params,
                        ds=ds,
                        laps=laps,
                        regen_val=r,
                        delta_val=d,
                        min_e_target=min_e,
                        max_e_target=max_e,
                        map_data=m_data
                    )
                )
            )

            # connect progress updates
            self.controller.simulationProgress.connect(win.loading_overlay.update_progress)
            self.controller.simulationFinished.connect(win.stop_loading)
            self.controller.simulationError.connect(win.stop_loading)
            
            self._register_and_show_window(win)
        except Exception as e:
            QMessageBox.critical(self.main_window, "Error", f"Could not load laptime map:\n{e}")

    def open_compare_window(self):
        from src.ui.results_tab import CompareWindow
        win = CompareWindow()
        self._register_and_show_window(win)

    # ==========================================
    # handling the results
    # ==========================================
    def handle_simulation_finished(self, payload):
        kind = payload.kind
        data = payload.data
        meta = payload.meta
    
        if kind == "single_lap":
            win = ResultsWindow(data, meta['car'], meta['track'], meta['ds'])
            self._register_and_show_window(win)

        elif kind == "endurance":
            win = EnduranceResultsWindow(data, meta['car'], meta['track'], meta['ds'])
            self._register_and_show_window(win)
        
        elif kind in ("endurance_map", "endurance_deployment"):
    
            for win in self.open_windows:
                if isinstance(win, LaptimeAppMainWindow):
                    win.add_result_tab(payload)
                    break

    def handle_error(self, err_msg):
        
        QMessageBox.critical(self.main_window, "Simulation error", str(err_msg))

    def _register_and_show_window(self, window):
        
        self.open_windows.append(window)
    
        window.setAttribute(Qt.WidgetAttribute.WA_DeleteOnClose)
        window.destroyed.connect(lambda: self.open_windows.remove(window) if window in self.open_windows else None)
        window.show()

from dataclasses import dataclass,field
import numpy as np
import casadi as ca


@dataclass
class Car:
    # --- 1. Parameters from UI ---
    weight: float           # [kg]
    cd: float             # [-] 
    cl: float             # [-]
    cd_drs: float 
    cl_drs: float
    acp_capacity: float
    drs_allowed: bool
    power_max: float 
    regen_eff: float 
    soc_start: float 
    cop: float 
    energy_limit: float = None  # [kWh] 
    performance_delta: float = 0.0

    # Calculated fields 
    fz_static_f: float = field(init=False)
    fz_static_r: float = field(init=False)
    cs_static_f : float = field(init=False)
    cs_static_r : float = field(init=False)

    model_type: str = "mass_point"

    # Constants
    rho = 1.225
    g = 9.81

    # Powertrain
    brake_max = 60000
    power_eff = 0.88
    fix_drag = 230
    wheel_radius = 0.205
    max_torque = 30.0
    gear_ratio = 11.46
    brake_bias = 0.55
    power_dist = 0.5
    
    #grip
    mu_x: float = 1.25           # [-] 
    mu_y: float = 1.25           # [-] 
    

    #soc limitations
    soc_grid_power = [0.0, 0.04, 0.08, 0.12, 0.22, 0.32, 0.42, 0.52, 0.70, 1.00,1.1]
    power_limit_W = [10e3, 14e3, 21e3, 35e3, 62e3, 79e3, 89e3, 95e3, 100e3, 100e3,100e3]
    soc_to_power_interp = ca.interpolant('soc_power', 'bspline', [soc_grid_power], power_limit_W)

    # Suspension
    cogr = 0.5 
    cogz = 0.3
    sus_t = 0.2
    Iz = 110 
    wheelbase = 1.53 
    weight_distrib = 0.46 # to rear
    steeringAngle_max = 25/180*ca.pi;
    
  
    lf = wheelbase * cogr
    lr = wheelbase - lf
    
    tire_model = { 
        "cold_factor": 0.8, 
        "warmup_tau": 1.6, 
        "noise_std": 0.03, 
        "mu_x_nominal": 1.22,
        "mu_y_nominal":1.22
    }

    def __post_init__(self):
        # calculate static weight
        self.fz_static_f = self.weight * self.g * (1 - self.weight_distrib)
        self.fz_static_r = self.weight * self.g * self.weight_distrib
        #self.cs_static_f = self.get_cornering_stiffness(self.fz_static_f)
        #self.cs_static_r = self.get_cornering_stiffness(self.fz_static_r)
        #print(f"cornering stiffness front: {self.cs_static_f}")
        self.cs_static_f = 2 * self.get_cornering_stiffness(self.fz_static_f/2)
        self.cs_static_r = 2 * self.get_cornering_stiffness(self.fz_static_r/2)


    # car calculation functions 
    def get_aero_drag(self, v, cd = None):
        #if we pass cd as optimization parameter
        use_cd = cd if cd is not None else self.cd
        return self.fix_drag + 0.5 * self.rho * use_cd * v**2
    

    def get_recuperation_limit(self, SoC_ca):
        """
        Vrací spojitý (hladký) limit rekuperace ve Wattech pro dané SoC.
        """
        P_max = 40e3   # 40 kW limit
        soc_mid = 0.885 # Kde křivka protíná 20 kW
        k = 120.0       # Strmost poklesu
        
        return P_max / (1.0 + ca.exp(k * (SoC_ca - soc_mid)))
    
    def tire_fy(self, alpha, Fz):
            """
            Linear tire model
            Fy = (alpha/pi*180)/5 * Fz * tire_fyFactor
            """
            return (alpha * 180.0 / ca.pi) / 5.0 * Fz * self.mu_y
    
    def split_long_forces(self, F_acc, F_brake):
        """
        Inputs:
          F_acc   >= 0
          F_brake <= 0
        Returns:
          Ff, Fr  (front/rear longitudinal forces)
        """
        Ff_acc = self.power_dist * F_acc
        Fr_acc = (1.0 - self.power_dist) * F_acc

        Ff_brk = self.brake_bias * F_brake
        Fr_brk = (1.0 - self.brake_bias) * F_brake

        Ff = Ff_acc + Ff_brk
        Fr = Fr_acc + Fr_brk

        return Ff, Fr
    
        
    def get_friction_limit(self, v, weight=None, use_cl=None):
        weight = weight if weight is not None else self.weight
        use_cl = use_cl if use_cl is not None else self.cl

        F_aero = 0.5 * self.rho * use_cl * v**2

        Fz = weight * self.g + F_aero
        
        # OPRAVA: Tady musí být násobení self.mu_x a self.mu_y, nikoliv weight!
        max_x = self.mu_x * Fz
        max_y = self.mu_y * Fz
        
        return (max_x, max_y)
    

    def get_fz(self,v,cla_ef):
        #for single track, returns Fz on front and rear
        fz_aero = 0.5*self.rho*cla_ef*v**2
        fz_aero_front = fz_aero*(1-self.cop)
        fz_aero_rear = fz_aero*self.cop

        fz_f = self.fz_static_f + fz_aero_front
        fz_r = self.fz_static_r + fz_aero_rear

        return fz_f,fz_r
    
    def get_tire_dy(self,fx,fz):

        #save option
        #term = (self.mu_y * fz)**2 - fx**2
        #term_safe = ca.log(1 + ca.exp(term))
        #dy =ca.sqrt(term_safe)
        max_lateral_sq = (self.mu_y * fz)**2 - fx**2
        # Použijeme fmax, aby pod odmocninou bylo minimálně 1e-6
        dy = ca.sqrt(ca.fmax(max_lateral_sq, 1))
        return dy

        #dy = ca.sqrt((self.mu_y*fz)**2 -fx**2) 
        #return dy
    
    def get_cornering_stiffness(self, fz):
       
        PKY1 = 19.7276 
        PKY2 = 3.3304
        PKY4 = 7.5668
        FNOMIN = 2000  # Z tir souboru

        # Pacejka vzorec pro Stiffness (bez Camberu)
        # C_alpha = PKY1 * FNOMIN * sin(PKY4 * arctan(Fz / (PKY2 * FNOMIN)))
        
        term_inner = fz / (PKY2 * FNOMIN)
        c_alpha = PKY1 * FNOMIN * ca.sin(PKY4 * ca.arctan(term_inner))
    
        return c_alpha
    
    
    def update_regen_usage(self, base_drift=-0.0015, sigma=0.02):
    
        noise = np.random.normal(0.0, sigma)
        self.regen_eff += np.round(base_drift + noise,2)
        self.regen_eff = round(np.clip(self.regen_eff, 0.5, 1.0),2)

    def update_grip(self, lap):
        
        #tire warmup
        warmup_progress = 1 - np.exp(-(lap + 1) / self.tire_model["warmup_tau"])

        mu_x_cold = self.tire_model["cold_factor"] * self.tire_model["mu_x_nominal"]
        current_base_mu_x = mu_x_cold + (self.tire_model['mu_x_nominal'] - mu_x_cold) * warmup_progress
        
        # Y složka
        mu_y_cold = self.tire_model["cold_factor"] * self.tire_model["mu_y_nominal"]
        current_base_mu_y = mu_y_cold + (self.tire_model['mu_y_nominal'] - mu_y_cold) * warmup_progress

        noise = self.tire_model["noise_std"] * np.random.randn()
      
        self.mu_x = np.round(current_base_mu_x + noise, 3)
        self.mu_y = np.round(current_base_mu_y + noise, 3)

     
        print(f"Lap {lap+1} Grip: Mu_X={self.mu_x}, Mu_Y={self.mu_y} (Warmup: {warmup_progress:.1%})")
        
       # return self.mu_x, self.mu_y

    def singletrack_time_ode(self, z, u, cl_eff=None, cd_eff=None):
            """
            Time-domain bicycle ODE.

            State:
            z = [vx, vy, psi, dpsi, X, Y]

            Input:
            u = [delta_f, F_acc, F_brake]

            Returns:
            dz/dt (CasADi expression)
            """

            vx   = z[0]
            vy   = z[1]
            psi  = z[2]
            r    = z[3]
            X    = z[4]
            Y    = z[5]

            delta_f = u[0]
            F_acc   = u[1]
            F_brake = u[2]

            # effective aero params 
            if cl_eff is None:
                cl_eff = self.cl
            if cd_eff is None:
                cd_eff = self.cd  

            # distribute long forces
            Ff, Fr = self.split_long_forces(F_acc, F_brake)

            # normal loads 
            Fz_f, Fz_r = self.get_fz(vx, cl_eff)

            # slip angles 
            alpha_f = -ca.atan2(
                (vy + self.lf * r) * ca.cos(delta_f) - vx * ca.sin(delta_f),
                (vy + self.lf * r) * ca.sin(delta_f) + vx * ca.cos(delta_f)
            )

            alpha_r = -ca.atan2(
                (vy - self.lr * r), vx
            )

            # lateral tire forces
            F_fy = self.tire_fy(alpha_f, Fz_f)
            F_ry = self.tire_fy(alpha_r, Fz_r)

            # aero drag (uses vx only, cd is CdA)
            F_drag = 0.5 * self.rho * cd_eff * vx**2 + self.fix_drag

            # dynamics (Matlab style)
            dvx = (Ff * ca.cos(delta_f) - F_fy * ca.sin(delta_f) + Fr - F_drag) / self.mass + r * vy
            dvy = (Ff * ca.sin(delta_f) + F_fy * ca.cos(delta_f) + F_ry) / self.mass - r * vx

            dpsi = r
            dr   = (self.lf * (Ff * ca.sin(delta_f) + F_fy * ca.cos(delta_f)) - self.lr * F_ry) / self.Iz

            dX = vx * ca.cos(psi) - vy * ca.sin(psi)
            dY = vx * ca.sin(psi) + vy * ca.cos(psi)

            return ca.vertcat(dvx, dvy, dpsi, dr, dX, dY)
    
    def singletrack_path_ode(self, s, z, u, track, cl_eff=None, cd_eff=None):
        """
        Path-domain bicycle ODE.

        Independent variable: s

        State:
          z = [vx, vy, psi, dpsi, n, t]

        Input:
          u = [delta_f, F_acc, F_brake]

        Track must provide interpolation:
          th(s) = track heading
          C(s)  = curvature

        Returns:
          dz/ds
        """

        vx   = z[0]
        vy   = z[1]
        psi  = z[2]
        r    = z[3]
        n    = z[4]
        t    = z[5]

        delta_f = u[0]
        F_acc   = u[1]
        F_brake = u[2]

        # effective aero params
        if cl_eff is None:
            cl_eff = self.cl
        if cd_eff is None:
            cd_eff = self.cd  # CdA

        # distribute long forces
        Ff, Fr = self.split_long_forces(F_acc, F_brake)

        # track geometry 
        th = track.heading(s)     # == track.psi(s)
        C  = track.curvature(s)   # == track.kappa(s)

        eps = psi - th

        
        denom = (vx * ca.cos(eps) - vy * ca.sin(eps))
        Sf = (1.0 - n * C) / denom

        # normal loads depend on vx only
        Fz_f, Fz_r = self.get_fz(vx, cl_eff)

        # slip angles 
        alpha_f = -ca.atan2(
            (vy + self.lf * r) * ca.cos(delta_f) - vx * ca.sin(delta_f),
            (vy + self.lf * r) * ca.sin(delta_f) + vx * ca.cos(delta_f)
        )

        alpha_r = -ca.atan2(
            (vy - self.lr * r), vx
        )

        # lateral tire forces
        F_fy = self.tire_fy(alpha_f, Fz_f)
        F_ry = self.tire_fy(alpha_r, Fz_r)

        # aero drag
        F_drag = 0.5 * self.rho * cd_eff * vx**2 + self.fix_drag

        # time-domain derivatives (same as time ODE)
        dvx_dt = (Ff * ca.cos(delta_f) - F_fy * ca.sin(delta_f) + Fr - F_drag) / self.mass + r * vy
        dvy_dt = (Ff * ca.sin(delta_f) + F_fy * ca.cos(delta_f) + F_ry) / self.mass - r * vx
        dpsi_dt = r
        dr_dt   = (self.lf * (Ff * ca.sin(delta_f) + F_fy * ca.cos(delta_f)) - self.lr * F_ry) / self.Iz

        # path coordinates dynamics
        dn_dt = vx * ca.sin(eps) + vy * ca.cos(eps)
        dt_dt = 1.0

        # convert dt derivatives to ds derivatives
        dvx_ds  = Sf * dvx_dt
        dvy_ds  = Sf * dvy_dt
        dpsi_ds = Sf * dpsi_dt
        dr_ds   = Sf * dr_dt
        dn_ds   = Sf * dn_dt
        dt_ds   = Sf * dt_dt

        return ca.vertcat(dvx_ds, dvy_ds, dpsi_ds, dr_ds, dn_ds, dt_ds)

    power: np.ndarray = field(repr=False) # kW
  
    length: float = 0.0

    kappa_interp: ca.Function = field(repr=False, default=None)

    @property
    def ref_data(self):
        """interface for validation module"""
        if self.speed is None:
            return None
            
        return {
            's': self.s,          # Vzdálenost (Osa X reference)
            'v': self.speed,      # Rychlost [m/s]
            'p': self.power       # Výkon [kW]
        }

    @classmethod
    def load(cls, track_name, tracks_dir="data/tracks"):
        """
        loads the .npz track for the simulation
        """
        filepath = os.path.join(tracks_dir, f"{track_name}.npz")
        
        if not os.path.exists(filepath):
            raise FileNotFoundError(f"Baked track '{track_name}' not found. Run baking first.")
            
        
        data = np.load(filepath)
        
        s, x, y = data['s'], data['x'], data['y']
        kappa = data['kappa']
        drs = data['drs'] if 'drs' in data else None
        
        psi = data['psi'] if 'psi' in data else None
        z = data['z'] if 'z' in data else None
        alpha = data['alpha'] if 'alpha' in data else None
        speed = data['speed'] if 'speed' in data else None
        brake = data['brake'] if 'brake' in data else None
        power = data['power'] if 'power' in data else None
       
       
        kappa_func = ca.interpolant('KappaProfile', 'bspline', [s], kappa)
        if psi is not None:
            psi_func = ca.interpolant('PsiProfile','linear',[s],psi)
        else:
            psi_func = None

        drs_func = None
        if drs is not None:
            
            drs_func = ca.interpolant('DRSProfile', 'linear', [s], drs)
        
        return cls(
            name=track_name,
            s=s, 
            x=x, 
            y=y, 
            z=z,
            drs=drs,
            psi=psi, 
            kappa=kappa,
            alpha=alpha,
            speed = speed,
            brake= brake,
            power = power,
            length=s[-1],
            kappa_interp=kappa_func,
            psi_interp = psi_func,
            drs_interp=drs_func
        )
    
   

    def get_curvature(self, s_var):
        if self.kappa_interp is None:
            raise RuntimeError("Interpolant not initialized.")
        return self.kappa_interp(s_var)
    
    def get_heading(self,s_var):
        if self.psi_interp is None:
            raise RuntimeError("Interpolant not initialized.")
        return self.psi_interp(s_var)
    
    def get_drs_flag(self, s_var):
        
        if self.drs_interp is None:
            # Pokud trať nemá DRS, vracíme 0 (funguje pro scalar i CasADi vector)
            if isinstance(s_var, (ca.MX, ca.SX)):
                return ca.DM.zeros(s_var.shape)
            return 0.0
            
        return self.drs_interp(s_var)

import numpy as np
import scipy.io

class TrackGenerator:
    @staticmethod
    def smooth_track(x_smpl, y_smpl, z_smpl, ds=0.05, r=1000, is_closed=True):
        x_smpl = np.array(x_smpl).flatten()
        y_smpl = np.array(y_smpl).flatten()
        z_smpl = np.array(z_smpl).flatten()
        
        dx = np.diff(x_smpl)
        dy = np.diff(y_smpl)
        ds_tmp = np.sqrt(dx**2 + dy**2)
        s_tmp = np.concatenate(([0], np.cumsum(ds_tmp)))
        
        total_length = s_tmp[-1]
        N = int(np.round(total_length / ds))
        if N < 2: N = 2 
        ds_real = total_length / N
        s_traj = np.linspace(0, total_length, N) 
        
        # Initial Interpolation
        fx = PchipInterpolator(s_tmp, x_smpl)
        fy = PchipInterpolator(s_tmp, y_smpl)
        fz = PchipInterpolator(s_tmp, z_smpl)
        x_init = fx(s_traj)
        y_init = fy(s_traj)
        z_init = fz(s_traj)
        
        # Init alpha, theta, C
        dz_init = np.diff(z_init) / ds_real
        alpha_init = np.arcsin(np.clip(dz_init, -0.2, 0.2))
        alpha_init = np.append(alpha_init, alpha_init[-1])

        th_init = np.unwrap(np.arctan2(np.diff(y_init), np.diff(x_init)))
        th_init = np.append(th_init, th_init[-1])
        C_init = np.append(np.diff(th_init) / ds_real, 0)
        
        # --- CasADi Optimization ---
        opti = ca.Opti()
        
        # State: [kappa, theta, x, y, alpha, z]
        Z = opti.variable(N, 6) 
        z_C, z_th, z_x, z_y, z_alp, z_z = Z[:,0], Z[:,1], Z[:,2], Z[:,3], Z[:,4], Z[:,5]
        
        # Control: [d_kappa/ds, d_alpha/ds]
        u = opti.variable(N-1, 2) 
        
        # Cost Function
        x_dev = ((z_x[:-1] - x_init[:-1])**2 + (z_x[1:] - x_init[1:])**2) / 2
        y_dev = ((z_y[:-1] - y_init[:-1])**2 + (z_y[1:] - y_init[1:])**2) / 2
        z_dev = ((z_z[:-1] - z_init[:-1])**2 + (z_z[1:] - z_init[1:])**2) / 2
        
        reg_term = r * u[:,0]**2 + 10*r * u[:,1]**2
        
        cost = ca.sum1(ds_real * (reg_term + x_dev + y_dev + z_dev))
        opti.minimize(cost)
        
        # Dynamics
        def f_dyn(z_in, u_in):
            return ca.vertcat(u_in[0],              # dC/ds
                              z_in[0],              # dth/ds = kappa
                              ca.cos(z_in[1]),      # dx/ds = cos(theta)
                              ca.sin(z_in[1]),      # dy/ds = sin(theta)
                              u_in[1],              # dalp/ds
                              ca.sin(z_in[4]))      # dz/ds = sin(alpha)

        for k in range(N-1):
            z_k, u_k = Z[k, :].T, u[k, :].T
            k1 = f_dyn(z_k, u_k)
            k2 = f_dyn(z_k + ds_real/2*k1, u_k)
            k3 = f_dyn(z_k + ds_real/2*k2, u_k)
            k4 = f_dyn(z_k + ds_real*k3, u_k)
            opti.subject_to(Z[k+1, :].T == z_k + ds_real/6 * (k1 + 2*k2 + 2*k3 + k4))

        # Constraints
        if is_closed:
            opti.subject_to(Z[-1, 2] == Z[0, 2]) # x
            opti.subject_to(Z[-1, 3] == Z[0, 3]) # y
            opti.subject_to(Z[-1, 5] == Z[0, 5]) # z
            opti.subject_to(Z[-1, 0] == Z[0, 0]) # kappa
            opti.subject_to(Z[-1, 4] == Z[0, 4]) # alpha
            n_loops = np.round((th_init[-1] - th_init[0]) / (2*np.pi))
            opti.subject_to(z_th[-1] == z_th[0] + 2*np.pi*n_loops)

        # Init guess
        opti.set_initial(z_x, x_init)
        opti.set_initial(z_y, y_init)
        opti.set_initial(z_z, z_init)
        opti.set_initial(z_th, th_init)
        opti.set_initial(z_C, C_init)
        opti.set_initial(z_alp, alpha_init)
        
        opti.solver('ipopt', {'expand': True}, {'max_iter': 500, 'print_level': 0})
        
        try:
            sol = opti.solve()
            Z_opt = sol.value(Z)
            return {
                's': s_traj, 'kappa': Z_opt[:, 0], 'psi': Z_opt[:, 1], 
                'x': Z_opt[:, 2], 'y': Z_opt[:, 3], 'alpha': Z_opt[:, 4], 'z': Z_opt[:, 5]
            }
        except Exception as e:
            print(f"Optimalizace selhala: {e}")
            return {'s': s_traj, 'kappa': C_init, 'psi': th_init, 'x': x_init, 'y': y_init, 'alpha': alpha_init, 'z': z_init}

        x_dev = w_pos*((z_x[:-1] - x_init[:-1])**2 + (z_x[1:] - x_init[1:])**2) / 2

        # --- OMEZENÍ (Constraints) ---
        if is_closed:
            opti.subject_to(Z[-1, 2] == Z[0, 2]) # x
            opti.subject_to(Z[-1, 3] == Z[0, 3]) # y
            opti.subject_to(Z[-1, 5] == Z[0, 5]) # z
            opti.subject_to(Z[-1, 0] == Z[0, 0]) # kappa
            opti.subject_to(Z[-1, 4] == Z[0, 4]) # alpha
            n_loops = np.round((th_init[-1] - th_init[0]) / (2*np.pi))
            opti.subject_to(z_th[-1] == z_th[0] + 2*np.pi*n_loops)

        # Inicializace
        opti.set_initial(z_x, x_init)
        opti.set_initial(z_y, y_init)
        opti.set_initial(z_z, z_init)
        opti.set_initial(z_th, th_init)
        opti.set_initial(z_C, C_init)
        opti.set_initial(z_alp, alpha_init)
        
        # Nastavení solveru (stejně jako v Matlabu)
        opti.solver('ipopt', {'expand': True}, {'max_iter': 500, 'print_level': 0})
        
        try:
            sol = opti.solve()
            Z_opt = sol.value(Z)
            return {
                's': s_traj, 'kappa': Z_opt[:, 0], 'psi': Z_opt[:, 1],

            return {'s': s_traj, 'kappa': C_init, 'psi': th_init, 'x': x_init, 'y': y_init, 'alpha': alpha_init, 'z': z_init}

    @staticmethod
    def get_tangent_vector(data, latitude_ref, longitude_ref):
        lon, lat = data["Longitude"], data["Latitude"]
        distances = np.sqrt((lon - longitude_ref)**2 + (lat - latitude_ref)**2)
        closest_idx = np.argmin(distances)
        vector = np.array([0.0, 0.0])
        for i in range(1, 51):
            if closest_idx + i < len(lon):
                vector = np.array([lon[closest_idx + i] - lon[closest_idx],
                                lat[closest_idx + i] - lat[closest_idx]])
                if not np.allclose(vector, [0, 0]): break
        return closest_idx, vector

    @staticmethod
    def get_lap_indices(data, point, tangent_vector):
        lat, lon = data["Latitude"], data["Longitude"]
        rel_long, rel_lat = lon - point[1], lat - point[0]
        vec_norm = np.dot(tangent_vector, tangent_vector)
        projected = np.array([np.dot([rel_long[i], rel_lat[i]], tangent_vector) / vec_norm for i in range(len(lon))])
        minima_indices, _ = find_peaks(-np.abs(projected))
        return [i for i in minima_indices if (abs(rel_long[i]) + abs(rel_lat[i])) < 0.0002]

    @staticmethod
    def process_and_save_lap(all_data, lap_info, is_closed, save_path):
        s, e = lap_info['indices']
        
        # 1. Extrakce surových dat
        raw_lat = all_data['Latitude'][s:e]
        raw_lon = all_data['Longitude'][s:e]
        raw_alt = all_data['Altitude'][s:e]
        raw_speed = all_data['Speed'][s:e]
        raw_power = all_data['Power'][s:e]
        raw_brake = all_data['Brake'][s:e]

        # 2. Příprava masek
        # A) Speed Mask
        speed_mask = raw_speed > 1.0
        
        # B) Distance Mask (převod na metry pro výpočet vzdáleností)
        x_m_temp, y_m_temp = TrackGenerator.latlon_to_meters(raw_lat, raw_lon)
        dx = np.diff(x_m_temp, prepend=x_m_temp[0])
        dy = np.diff(y_m_temp, prepend=y_m_temp[0])
        dist_sq = dx**2 + dy**2
        dist_mask = np.concatenate(([True], dist_sq[1:] > 1e-6)) # 1mm tolerance

        # 3. Aplikace kombinované masky na VŠECHNA DATA
        final_mask = speed_mask & dist_mask
        
        # DŮLEŽITÉ: Musíme zkrátit všechna pole, aby seděla délka
        raw_lat = raw_lat[final_mask]
        raw_lon = raw_lon[final_mask]
        raw_alt = raw_alt[final_mask]
        raw_speed = raw_speed[final_mask]
        raw_power = raw_power[final_mask]
        raw_brake = raw_brake[final_mask]

        # Znovu přepočítáme x_m, y_m už z vyfiltrovaných Lat/Lon
        # (Nebo bychom mohli filtrovat x_m_temp, ale takhle je to čistší)
        x_m, y_m = TrackGenerator.latlon_to_meters(raw_lat, raw_lon)

        if len(x_m) < 10:
            msg = "Error: Not enough valid data points. Car must be strictly in motion (> 1 m/s)."
            return False, msg # <--- VRACÍ TUPLE
        
        # 4. Smoothing
        tg = TrackGenerator()
        smoothed = tg.smooth_track(x_m, y_m, raw_alt, ds=0.1, r=1e-3, is_closed=is_closed)
        
        if smoothed is None: 
            return False, "Smoothing optimization failed (IPOPT error)." # <--- VRACÍ TUPLE
        
        # 5. Interpolace Telemetrie
        # s_raw počítáme z FILTROVANÝCH dat (x_m, y_m mají nyní stejnou délku jako raw_speed atd.)
        s_raw = np.concatenate(([0], np.cumsum(np.sqrt(np.diff(x_m)**2 + np.diff(y_m)**2))))
        
        def interp_on_s(data_arr):
            # data_arr musí mít stejnou délku jako s_raw!
            f = interp1d(s_raw, data_arr, kind='linear', fill_value="extrapolate")
            return f(smoothed['s'])

        final_data = {
            's': smoothed['s'],
            'x': smoothed['x'],
            'y': smoothed['y'],
            'z': smoothed['z'],
            'kappa': smoothed['kappa'],
            'psi': smoothed['psi'],
            'alpha': smoothed['alpha'],
            'speed': interp_on_s(raw_speed), 
            'power': interp_on_s(raw_power),
            'brake': interp_on_s(raw_brake),
        }
        
        np.savez(save_path, **final_data)
        
        return True, "Track generated successfully."

class SimulationController(QObject):
    # central signals for ui updates
    simulationStarted = pyqtSignal()
    simulationProgress = pyqtSignal(int, int)
    simulationFinished = pyqtSignal(object) 
    simulationError = pyqtSignal(str)

    def __init__(self):
        super().__init__()
        self.active_worker = None

    def _create_solver(self, car: Car, track: Track, laps: int):
        # helper method to build the right solver based on car configuration
        if car.model_type == "single_track":
            return SingleTrackSolver(car, track, laps=laps)
        elif car.model_type == "mass_point_rk4":
            return MassPointSolver(car, track, rk4=True, laps=laps)
        else:
            return MassPointSolver(car, track, rk4=False, laps=laps)

    def handle_request(self, req):
        # single entry point for all incoming simulation requests
        self.simulationStarted.emit()

        try:
            # build the core car object shared across all requests
            car = Car(**req.car_params)
            task = None

            # route logic based on the specific type of request
            if isinstance(req, SingleLapRequest):
                track = Track.load(req.track_name)
                solver = self._create_solver(car, track, laps=1)
                task = SingleLapTask(solver, car, track, req.ds)
                
            elif isinstance(req, EnduranceRequest):
                track = Track.load(req.track_name)
                solver = self._create_solver(car, track, laps=req.laps)
                # task = EnduranceTask(solver, car, track, req.ds, req.laps)
                raise NotImplementedError("endurance task not implemented yet")

            elif isinstance(req, LapMapTestRequest):
                track = Track.load(req.track_name)
                solver = self._create_solver(car, track, laps=req.laps)
                task = LaptimeMapTask(solver, car, track, req.ds, req.laps)

            elif isinstance(req, (StrategyOptRequest, DeploymentOptRequest)):
                # update car with strategy specific modifiers
                car.regen_eff = req.regen_val
                car.performance_delta = req.delta_val
                
                laptime_interpolant = req.map_data['interpolant']
                solver = StrategyMaker(
                    car, laptime_interpolant, req.laps, 
                    req.min_e_target + 0.01, req.max_e_target - 0.01
                )
                
                if type(req) is StrategyOptRequest:
                    task = StrategyOptTask(solver, car)
                elif type(req) is DeploymentOptRequest:
                    task = DeploymentOptTask(solver, car)
            else:
                # catch unknown requests
                raise TypeError(f"unknown request type: {type(req).__name__}")

            # delegate the task to the worker thread
            self.active_worker = SimulationWorker(task)
            
            # connect signals
            self.active_worker.progress.connect(self.simulationProgress.emit)
            self.active_worker.error.connect(self._on_worker_error)
            self.active_worker.finished.connect(self.simulationFinished.emit)
            self.active_worker.finished.connect(self._cleanup_worker)

            # start computation
            self.active_worker.start()

        except Exception as e:
            self.simulationError.emit(f"failed to initialize task: {str(e)}")

    def _on_worker_error(self, err_msg):
        # emit error and clean up
        self.simulationError.emit(err_msg)
        self._cleanup_worker()

    def _cleanup_worker(self):
        # safely delete the c++ thread object
        if self.active_worker:
            self.active_worker.deleteLater()
            self.active_worker = None

        # =========================
        p_weight = opti.parameter()
        p_cl   = opti.parameter()
        p_cd   = opti.parameter()
 

        # Set params values
        opti.set_value(p_weight, self.car.weight)
        opti.set_value(p_cl,   self.car.cl)
        opti.set_value(p_cd,   self.car.cd)
   

        # =========================
        # Track data
        # =========================
        total_length = self.track.length
        N = int(np.round(total_length / ds))

        s_sim = np.linspace(0, total_length, N + 1)
        ds = s_sim[1:] - s_sim[:-1]

        kappa_grid = self.track.get_curvature(s_sim)

        # =========================
        # States
        # =========================
        v = opti.variable(N + 1)   # speed [m/s]
        T = opti.variable(N + 1)   # time [s]
        E = opti.variable(N + 1)   # energy used [kWh]
        
        # =========================
        # Inputs
        # =========================
        F_acc   = opti.variable(N)   # Gas [N]
        F_brake_mech = opti.variable(N)
        F_brake_regen = opti.variable(N)
        F_brake = F_brake_mech + F_brake_regen
        drs     = opti.variable(N)

        
        # =========================
        # DRS
        # =========================
        if self.car.drs_allowed and self.track.drs is not None:
            drs_flag = self.track.get_drs_flag(s_sim[:-1])
            drs_ef = drs * drs_flag
            cl_eff = p_cl + drs_ef * (self.car.cl_drs - p_cl)
            cd_eff = p_cd + drs_ef * (self.car.cd_drs - p_cd)
        else:
            drs_ef = ca.DM.zeros(N)
            cl_eff = p_cl
            cd_eff = p_cd

        opti.subject_to(drs >= 0)
        opti.subject_to(drs <= 1)

        # Current speed (k)
        v_k = v[:-1]

        # =========================
        # Aero
        # =========================
        F_drag = Car.get_aero_drag(self.car, v_k, cd=cd_eff)

        F_max_x, F_max_y = Car.get_friction_limit(
            self.car, v_k,
            weight=p_weight,
            use_cl=cl_eff
        )

        # Lateral Force: F_lat = m * v^2 * curvature
        F_lat = p_weight * v_k**2 * ca.fabs(kappa_grid[:-1])

        # Longitudinal force
        F_long_k = F_acc + F_brake

        # =========================
        # Pedal constraints
        # =========================
        opti.subject_to(F_acc >= 0)
        opti.subject_to(F_brake_regen <= 0)
        opti.subject_to(F_brake_mech <= 0)

        # =========================
        # Friction ellipse
        # =========================
        term_long = (F_long_k / F_max_x)**2
        term_lat  = (F_lat    / F_max_y)**2
        opti.subject_to(term_long + term_lat <= 1)

        #SoC
        E_left = self.car.acp_capacity - E[:-1]
        SoC = E_left/self.car.acp_capacity

        # =========================
        # Dynamics
        # =========================
        F_net = F_long_k - F_drag
        acc = F_net / p_weight

        # Speed integration: v^2 = v0^2 + 2*a*ds
        v_next_sq = v_k**2 + 2 * acc * ds
        opti.subject_to(v[1:]**2 == v_next_sq)

        # Time integration: T = T0 + ds/v
        opti.subject_to(T[1:] == T[:-1] + ds / v_k)

        # Recuperation limit at given SoC
        recuperation_max = self.car.get_recuperation_limit(SoC)

        P_recuperated = -F_brake_regen*v_k

        opti.subject_to(P_recuperated/recuperation_max <= 1)
        opti.subject_to((-F_brake_mech*v_k)/self.car.brake_max <= 1)
        opti.subject_to(F_brake/self.car.brake_max <= 1)
        
        #brakeP = -F_brake * v_k
        #P_recuperated = ca.fmin(brakeP, recuperation_max)

        energy_consumed = (F_acc * ds) / self.car.power_eff
        
        energy_saved = P_recuperated * self.car.regen_eff * (ds / v_k) * 0.94 #0.94 is fixed regen factor extracted from data
       
        dE_step = energy_consumed - energy_saved

        # Integrace do kWh
        opti.subject_to(E[1:] == E[:-1] + dE_step / 3.6e6)

        # =========================
        # Torque limit (30 Nm per wheel)
        # =========================
        opti.subject_to(
            ((self.car.wheel_radius * F_acc)) /
            (self.car.max_torque * self.car.gear_ratio * 4) < 1
        )

        # =========================
        # Power limit
        # =========================

        P_out = (F_acc * v_k) / self.car.power_eff

        # Limit from SoC (dynamic)
        P_max_dynamic = self.car.soc_to_power_interp(SoC)
        opti.subject_to(P_out <= P_max_dynamic)

        # Hard limit
        opti.subject_to(P_out <= self.car.power_max)

        # Bounding SoC
        opti.subject_to(SoC >= 0.0)

        # brake limit
        #opti.subject_to((brakeP * self.car.regen_eff) / self.car.brake_max <= 1)

        # Speed limits
        opti.subject_to(v >= 1.0)  # avoid division by zero

        # =========================
        # Initial conditions
        # =========================
        opti.subject_to(v[0] == init_cond.v)
        opti.subject_to(T[0] == init_cond.T)
        opti.subject_to(E[0] == init_cond.E)
        
        opti.subject_to(drs[0] == 0)  # start with DRS closed

        # =========================
        # Energy limitation
        # =========================
   
        if self.car.energy_limit is not None:
            opti.subject_to(E[-1]-init_cond.E <= self.car.energy_limit)

        # =========================
        # Objective function
        # =========================
        J_Time = T[-1]

        F_norm_factor = (self.car.weight * self.car.g)**2
        J_overlap = 0.004 * ca.sum1((F_acc * -F_brake) / F_norm_factor)

        J_final_v = 0.1 * v[-1]

        J_drs_binary = 0.002 * ca.sum1(drs * (1 - drs))

        opti.minimize(J_Time + J_overlap - J_final_v)

        # =========================
        # Initial guess
        # =========================
        opti.set_initial(v, 15.0)
        opti.set_initial(F_acc, 100.0)

        # =========================
        # Solver setup
        # =========================
        p_opts = {"expand": True}
        s_opts = {"max_iter": 2600, "tol": 1e-2, "print_level": 5}

        opti.solver("ipopt", p_opts, s_opts)

        try:
            sol = opti.solve()

            # =========================
            # Sensitivity analysis
            # =========================
            L = opti.f + ca.dot(opti.lam_g, opti.g)

            sens_w   = sol.value(ca.jacobian(L, p_weight))
            sens_cl  = sol.value(ca.jacobian(L, p_cl))
            sens_cd  = sol.value(ca.jacobian(L, p_cd))
         
            print("sensitivity results")
            print(f"Weight: {sens_w:+.4f} s/kg")
            print(f"CL:   {sens_cl:+.4f} s/unit")
            print(f"CdA:  {sens_cd:+.4f} s/unit")

            return SolverResults(
                success=True,
                lap_time=sol.value(T[-1]),
                t=sol.value(T),
                v=sol.value(v),
                s=s_sim,
                E=sol.value(E),
                SoC=sol.value(SoC)*100,
                f_acc=sol.value(F_acc) / self.car.power_eff,
                f_brake=sol.value(F_brake),
                recuperation=sol.value(P_recuperated*self.car.regen_eff),
                drs_usage=sol.value(drs_ef),
                sens={
                    "dTime_dm": sens_w,
                    "dTime_dCL": sens_cl,
                    "dTime_dCdA": sens_cd,
                }
            )

        except RuntimeError as e:
            print(f"Solver failed: {e}")
            return SolverResults(False, 0, [], [], [], [], [], [], {})
        
    def solveLap_rk4(self, init_cond: InitConditions, init_guess=None, ds=0.5):

        opti = ca.Opti()

        # Track data

        total_length = self.track.length
        N = int(np.round(total_length / ds))

        s_sim = np.linspace(0, total_length, N + 1)
        ds = s_sim[1:] - s_sim[:-1]

        v = opti.variable(N + 1)   # speed [m/s]
        T = opti.variable(N + 1)   # time [s]
        E = opti.variable(N + 1)   # energy used [kWh]

        F_acc   = opti.variable(N)   # Gas [N]

        F_brake_regen = opti.variable(N)

        # =========================

        if self.car.drs_allowed and self.track.drs is not None:

            drs_ef = drs * drs_flag
            cl_eff = p_cl + drs_ef * (self.car.cl_drs - p_cl)

        else:

            cd_eff = p_cd

        opti.subject_to(drs >= 0)
        opti.subject_to(drs <= 1)

        # Current speed (k)
        v_k = v[:-1]

        # =========================

        F_drag = Car.get_aero_drag(self.car, v_k, cd=cd_eff)

            self.car, v_k,

            use_cl=cl_eff
        )

        # Lateral Force: F_lat = m * v^2 * curvature

        # Longitudinal force
        F_long_k = F_acc + F_brake

        opti.subject_to(F_acc >= 0)

        opti.subject_to(F_brake/self.car.brake_max <= 1)

        # =========================

        # Pomocná funkce pro získání derivací stavů podle dráhy 's'
        def get_derivatives(v_val):
            # Odpor se průběžně mění s rychlostí uvnitř RK4 sub-kroků
            f_drag_step = Car.get_aero_drag(self.car, v_val, cd=cd_eff)
            f_net_step = F_long_k - f_drag_step
            acc_step = f_net_step / p_weight

        # =========================

        opti.subject_to(v[0] == init_cond.v)

class SingleTrackSolver:
    def __init__(self, car, track, laps=1):
        self.car = car
        self.track = track
        self.laps = laps

    def solveLap(self, init_cond: InitConditions,init_guess=None, ds=1.0):
        opti = ca.Opti()

        # =========================
        # Parameters
        # =========================
        p_mass = opti.parameter()
        p_cl   = opti.parameter()
        p_cd   = opti.parameter()
        p_mu_x = opti.parameter()
        p_mu_y = opti.parameter()

        # Set params values
        opti.set_value(p_mass, self.car.mass)
        opti.set_value(p_cl,   self.car.cl)
        opti.set_value(p_cd,   self.car.cd)
        opti.set_value(p_mu_x, self.car.mu_x)
        opti.set_value(p_mu_y, self.car.mu_y)

        # =========================
        # Track data & Grid
        # =========================
        total_length = self.track.length
        N = int(np.round(total_length / ds))
        
        s_sim = np.linspace(0, total_length, N + 1)
        ds_val = s_sim[1] - s_sim[0] # Actual uniform step
        
        # Curvature along the path
        kappa_grid = self.track.get_curvature(s_sim[:-1]) # Curvature at steps 0..N-1

        # =========================
        # Variables (States at k=0..N)
        # =========================
        # Note: We use N+1 points for states, N for controls
        vx  = opti.variable(N + 1) # Longitudinal speed
        vy  = opti.variable(N + 1) # Lateral speed
        r   = opti.variable(N + 1) # Yaw rate
        n   = opti.variable(N + 1) # Lateral deviation
        xi  = opti.variable(N + 1) # Heading error (psi - psi_track)
        t   = opti.variable(N + 1) # Time
        E   = opti.variable(N + 1) # Energy

        # =========================
        # Controls (at k=0..N-1)
        # =========================
        delta   = opti.variable(N) # Steering angle
        F_acc   = opti.variable(N) # Drive force (at wheels)
        F_brake = opti.variable(N) # Brake force (at wheels)
        drs     = opti.variable(N) # DRS State

        # =========================
        # Helper / Constants
        # =========================
        # Track limits (Tunnel)
        track_width_limit = 0.5 
        opti.subject_to(opti.bounded(-track_width_limit, n, track_width_limit))

        # Steering limits
        opti.subject_to(opti.bounded(-self.car.steeringAngle_max, delta, self.car.steeringAngle_max))

        # Speed limit (avoid singularity at 0)
        opti.subject_to(vx >= 1.0) 

        # Pedal constraints
        opti.subject_to(F_acc >= 0)
        opti.subject_to(F_brake <= 0)

        # DRS Logic
        use_drs_vec = False # Flag to control indexing inside loop

        if self.car.drs_allowed and self.track.drs is not None:
            drs_flag = self.track.get_drs_flag(s_sim[:-1])
            drs_ef = drs * drs_flag
            
            cl_eff = p_cl + drs_ef * (self.car.cl_drs - p_cl)
            cd_eff = p_cd + drs_ef * (self.car.cd_drs - p_cd)
            use_drs_vec = True
        else:
            drs_ef = ca.DM.zeros(N)
            cl_eff = p_cl # Scalar parameter
            cd_eff = p_cd # Scalar parameter
            opti.subject_to(drs == 0)
            use_drs_vec = False #

        opti.subject_to(drs >= 0)
        opti.subject_to(drs <= 1)

        # =========================
        # DYNAMICS LOOP (Euler Integration)
        # =========================
        
        # Slip limit for linear model validity (5 degrees in rad)
        alpha_limit = 5.0 * (np.pi / 180.0)

        for k in range(N):
            # Current states
            vx_k = vx[k]
            vy_k = vy[k]
            r_k  = r[k]
            n_k  = n[k]
            xi_k = xi[k]
            t_k  = t[k]
            
            # Current inputs
            delta_k   = delta[k]
            F_acc_k   = F_acc[k]
            F_brake_k = F_brake[k]
            
            curv_k = kappa_grid[k]

            # 1. Force distribution
            
            Ff_acc = self.car.power_dist * F_acc_k
            Fr_acc = (1.0 - self.car.power_dist) * F_acc_k
            
            Ff_brk = self.car.brake_bias * F_brake_k
            Fr_brk = (1.0 - self.car.brake_bias) * F_brake_k
            
            Fx_f = Ff_acc + Ff_brk
            Fx_r = Fr_acc + Fr_brk

            # 2. Normal loads (fz)
            
            if use_drs_vec:
                cl_k = cl_eff[k]
                cd_k = cd_eff[k]
            else:
                cl_k = cl_eff 
                cd_k = cd_eff

            # Fz calculations
            fz_aero = 0.5 * self.car.rho * cl_k * vx_k**2
            fz_aero_f = fz_aero * (1 - self.car.cop)
            fz_aero_r = fz_aero * self.car.cop
            
            Fz_f = self.car.fz_static_f + fz_aero_f
            Fz_r = self.car.fz_static_r + fz_aero_r

            # 3. Slip angles - kinematics
            # alpha_f = -atan2(...)
            # alpha_r = -atan2(...)
            
            # Front slip
            vy_f_kin = (vy_k + self.car.lf * r_k)
            alpha_f = -ca.atan2(
                vy_f_kin * ca.cos(delta_k) - vx_k * ca.sin(delta_k),
                vy_f_kin * ca.sin(delta_k) + vx_k * ca.cos(delta_k)
            )

            # Rear slip
            vy_r_kin = (vy_k - self.car.lr * r_k)
            alpha_r = -ca.atan2(vy_r_kin, vx_k)

            # 4. Lateral forces
            # Fy = (alpha_deg / 5.0) * Fz * mu_y
            # Convert alpha to deg: alpha * 180/pi
            # Formula: alpha_rad * (180/pi) / 5.0 * Fz * mu_y
            #        = alpha_rad * (36.0/pi) * Fz * mu_y
            
            Fy_f_lin = alpha_f * (36.0 / np.pi) * Fz_f * p_mu_y
            Fy_r_lin = alpha_r * (36.0 / np.pi) * Fz_r * p_mu_y

            # 5. constraints
            
            # A) Hard limit on slip angle to keep linear model valid
            opti.subject_to(opti.bounded(-alpha_limit, alpha_f, alpha_limit))
            opti.subject_to(opti.bounded(-alpha_limit, alpha_r, alpha_limit))

            # B) Friction ellipse 
            # (Fx / Fx_max)^2 + (Fy / Fy_max)^2 <= 1
            # Fx_max = mu_x * Fz, Fy_max = mu_y * Fz
        
            
            mu_x_fz_f = p_mu_x * Fz_f
            mu_y_fz_f = p_mu_y * Fz_f
            
            mu_x_fz_r = p_mu_x * Fz_r
            mu_y_fz_r = p_mu_y * Fz_r

            ellipse_f = (Fx_f / mu_x_fz_f)**2 + (Fy_f_lin / mu_y_fz_f)**2
            ellipse_r = (Fx_r / mu_x_fz_r)**2 + (Fy_r_lin / mu_y_fz_r)**2
            
            opti.subject_to(ellipse_f <= 1.0)
            opti.subject_to(ellipse_r <= 1.0)

            # 6. Power & Torque Limits
            # Torque: F_acc * radius / (ratio * 4) < max_torque (simplified per wheel check)
            # But F_acc is total force. 
            # Total Torque = F_acc * radius / ratio
            # Check against engine max torque * gear ratio logic or simplier force limit
            # Using your class params:
            
            # Power limit: P = F * v <= P_max
            opti.subject_to( ((F_acc_k * vx_k) / self.car.power_eff)/self.car.power_max <= 1 )
            
            # Regen limit
            opti.subject_to( ((-F_brake_k * vx_k) * self.car.regen_eff)/self.car.brake_max <= 1 )

            # Torque limit (simplified force limit at wheel)
            #max_force_torque = (self.car.max_torque * self.car.gear_ratio * 4) / self.car.wheel_radius
            #opti.subject_to( F_acc_k <= max_force_torque )

            # 7. Dynamics 
            # Based on singletrack_path_ode logic
            
            # Drag
            F_drag = 0.5 * self.car.rho * cd_k * vx_k**2 + self.car.fix_drag

            # Body frame forces
            # Sum forces in x and y direction
            F_x_total = Fx_f * ca.cos(delta_k) - Fy_f_lin * ca.sin(delta_k) + Fx_r - F_drag
            F_y_total = Fx_f * ca.sin(delta_k) + Fy_f_lin * ca.cos(delta_k) + Fy_r_lin
            
            # Moment
            M_z = self.car.lf * (Fx_f * ca.sin(delta_k) + Fy_f_lin * ca.cos(delta_k)) - self.car.lr * Fy_r_lin

            # Time derivatives (dt)
            dvx_dt = F_x_total / p_mass + r_k * vy_k
            dvy_dt = F_y_total / p_mass - r_k * vx_k
            dr_dt  = M_z / self.car.Iz
            
            # Path scaling factor (Sf = dt/ds)
            # d_psi/dt = r
            # d_theta/dt = vx * curvature? No, d_theta/ds = curvature -> d_theta/dt = ds/dt * curvature
            # We need Sf = 1 / (ds/dt)
            # s_dot = (vx * cos(xi) - vy * sin(xi)) / (1 - n * curvature)
            
            # eps (in code) is xi (here). 
            # Note: your code uses eps = psi - th. Here xi = psi - th.
            s_dot_denom = (1.0 - n_k * curv_k)
            s_dot_num   = (vx_k * ca.cos(xi_k) - vy_k * ca.sin(xi_k))
            
            # Avoid division by zero
            ds_dt = s_dot_num / ca.fmax(s_dot_denom, 0.1)
            Sf = 1.0 / ds_dt  # dt/ds

            # Spatial derivatives (d/ds = d/dt * dt/ds)
            dvx_ds = dvx_dt * Sf
            dvy_ds = dvy_dt * Sf
            dr_ds  = dr_dt  * Sf
            dt_ds  = Sf
            
            # Kinematics in path coordinates
            # dn/dt = vx * sin(xi) + vy * cos(xi)
            dn_ds = (vx_k * ca.sin(xi_k) + vy_k * ca.cos(xi_k)) * Sf
            
            # dxi/ds = dpsi/ds - dtheta/ds = r * Sf - curvature
            dxi_ds = r_k * Sf - curv_k

            # Energy
            dE_ds = ((F_acc_k / self.car.power_eff + F_brake_k * self.car.regen_eff) * Sf) / 3.6e6

            # Euler Step
            opti.subject_to(vx[k+1] == vx_k + dvx_ds * ds_val)
            opti.subject_to(vy[k+1] == vy_k + dvy_ds * ds_val)
            opti.subject_to(r[k+1]  == r_k  + dr_ds  * ds_val)
            opti.subject_to(n[k+1]  == n_k  + dn_ds  * ds_val)
            opti.subject_to(xi[k+1] == xi_k + dxi_ds * ds_val)
            opti.subject_to(t[k+1]  == t_k  + dt_ds  * ds_val)
            opti.subject_to(E[k+1] == E[k] + dE_ds * ds_val)


        # =========================
        # Boundary Conditions
        # =========================
        # Start
        opti.subject_to(vx[0] == init_cond.v)
        opti.subject_to(vy[0] == 0) # Assume 0 lateral speed at start
        opti.subject_to(r[0]  == 0) # Assume 0 yaw rate at start (or match curvature)
        opti.subject_to(n[0]  == init_cond.n)
        opti.subject_to(xi[0] == init_cond.xi)
        opti.subject_to(t[0]  == init_cond.T)
        opti.subject_to(E[0]  == init_cond.E)

        # Cyclic condition (if needed, enable for closed track)
        # opti.subject_to(vx[0] == vx[-1])
        
        # Energy limit
        if self.car.energy_limit is not None:
            opti.subject_to(E[-1] <= self.car.energy_limit)

        # =========================
        # Objective function
        # =========================
        J_Time = t[-1]
        
        # Regularization terms
        # 1. Steering Rate (smooth steering)
        # sum (delta_k - delta_k-1)^2
        diff_delta = delta[1:] - delta[:-1]
        J_steer_rate = 7e-4 * ca.sumsqr(diff_delta)
        
        # 2. Acceleration smoothness (throttle smoothing)
        diff_acc = F_acc[1:] - F_acc[:-1]
        J_acc_smooth = 1e-3 * ca.sumsqr(diff_acc) / 1000.0 # Scale down force magnitude

        # 3. Path deviation (stay close to center/opt line)
        J_path = 0.5 * ca.sumsqr(n)
        #0.5 funguje mega dobre

    
        opti.minimize(J_Time+J_path+J_steer_rate)
        #opti.minimize(J_Time)

        # =========================
        # Initial guess 
        # =========================
   
        # Mapping guess to variables
        # Note: guess arrays might be slightly different length due to float math, strict check needed
        g_vx = init_guess['v']
        g_n  = init_guess['n']
        g_xi = init_guess['xi']
        g_r  = init_guess['r']
        g_t  = init_guess['t']
        g_delta = init_guess['delta']
        g_facc  = init_guess['F_acc']
        g_fbrk  = init_guess['F_brake']

        # set initial (handling minimal length mismatch if any)
        L_state = min(len(g_vx), N + 1)
        opti.set_initial(vx[:L_state], g_vx[:L_state])
        opti.set_initial(vy[:L_state], 0.0) # Start with 0 slip
        opti.set_initial(n[:L_state],  g_n[:L_state])
        opti.set_initial(xi[:L_state], g_xi[:L_state])
        opti.set_initial(r[:L_state],  g_r[:L_state])
        opti.set_initial(t[:L_state],  g_t[:L_state])
        
        L_ctrl = min(len(g_delta), N)
        opti.set_initial(delta[:L_ctrl], g_delta[:L_ctrl])
        opti.set_initial(F_acc[:L_ctrl], g_facc[:L_ctrl])
        opti.set_initial(F_brake[:L_ctrl], g_fbrk[:L_ctrl])

        # =========================
        # Solver
        # =========================
        p_opts = {"expand": True}
        s_opts = {
            "max_iter": 3000, 
            "tol": 1e-3, 
            "print_level": 5, 
            "mu_strategy": "adaptive" 
        }
        opti.solver("ipopt", p_opts, s_opts)

        try:
            sol = opti.solve()
            
            # --- Post processing / Sensitivities ---
            L_lagrange = opti.f + ca.dot(opti.lam_g, opti.g)
            sens_m = sol.value(ca.jacobian(L_lagrange, p_mass))
            sens_cl = sol.value(ca.jacobian(L_lagrange, p_cl))
            
            # Calculate alphas for result (re-evaluate expressions)
            # We can't easily get them from 'sol' directly as they are intermediate expressions
            # so we assume user can recalc them from states if needed, or we compute approximate:
            res_vx = sol.value(vx)
            res_vy = sol.value(vy)
            res_r  = sol.value(r)
            res_delta = sol.value(delta)
            
            # Recompute alphas for output (k=0..N-1)
            # (Simplified re-calculation for visualization)
            res_alpha_f = []
            res_alpha_r = []
            for k in range(N):
                v_x, v_y, yaw_r, delt = res_vx[k], res_vy[k], res_r[k], res_delta[k]
                vy_f = v_y + self.car.lf * yaw_r
                af = -np.arctan2(vy_f * np.cos(delt) - v_x * np.sin(delt), vy_f * np.sin(delt) + v_x * np.cos(delt))
                vy_r = v_y - self.car.lr * yaw_r
                ar = -np.arctan2(vy_r, v_x)
                res_alpha_f.append(af)
                res_alpha_r.append(ar)

            return SolverResults(
                success=True,
                lap_time=sol.value(t[-1]),
                t=sol.value(t),
                v=sol.value(vx), # approx total speed
                s=s_sim,
                f_acc=sol.value(F_acc) / self.car.power_eff,
                f_brake=sol.value(F_brake) * self.car.regen_eff,
                delta=sol.value(delta),
                vx=res_vx,
                vy=res_vy,
                r=res_r,
                n=sol.value(n),
                xi=sol.value(xi),
                alpha_f=np.array(res_alpha_f),
                alpha_r=np.array(res_alpha_r),
                E=sol.value(E),
                drs_usage=sol.value(drs_ef),
                sens={"dTime_dm": sens_m, "dTime_dCL": sens_cl}
            )

        except RuntimeError as e:
            print(f"Solver failed: {e}")
            # Return partial results (debug values) if available
            return SolverResults(False, 0.0, [], [], [], [], [], [], {}, mu_x=self.car.mu_x, mu_y=self.car.mu_y)

class MPCC_Solver:
    def __init__(self,car,track,endurance=False,laps=1):

        self.car = car
        self.track = track
        self.laps = laps
        self.endurance = endurance
        
        if self.endurance:
            self.target_lap = car.acp_capacity/laps

    def solveLap(self, init_cond: InitConditions, init_guess=None, ds=0.5):

        opti = ca.Opti()

        # =========================
        # Parameters
        # =========================
        p_mass = opti.parameter()
        p_cl   = opti.parameter()
        p_cd   = opti.parameter()
        p_mu_x = opti.parameter()
        p_mu_y = opti.parameter()

        # Set params values
        opti.set_value(p_mass, self.car.mass)
        opti.set_value(p_cl,   self.car.cl)
        opti.set_value(p_cd,   self.car.cd)
        opti.set_value(p_mu_x, self.car.mu_x)
        opti.set_value(p_mu_y, self.car.mu_y)

        # =========================
        # Track data
        # =========================
        total_length = self.track.length
        N = int(np.round(total_length / ds))

        s_sim = np.linspace(0, total_length, N + 1)
        ds = s_sim[1:] - s_sim[:-1]

        kappa_grid = self.track.get_curvature(s_sim)

        # =========================
        # States
        # =========================
        v = opti.variable(N + 1)   # speed [m/s]
        T = opti.variable(N + 1)   # time [s]
        E = opti.variable(N + 1)   # energy used [kWh]
        n = opti.variable(N + 1)   # distance of centerline
        phi = opti.variable(N + 1) # heading error [rad]

        # =========================
        # Inputs
        # =========================
        F_acc   = opti.variable(N)   # Gas [N]
        F_brake = opti.variable(N)   # Brake [N]
        k_car   = opti.variable(N)   # Curvature of the car's path
        drs     = opti.variable(N)   # DRS position [0-1]

        # =========================
        # DRS
        # =========================
        if self.car.drs_allowed and self.track.drs is not None:
            drs_flag = self.track.get_drs_flag(s_sim[:-1])
            drs_ef = drs * drs_flag
            cl_eff = p_cl + drs_ef * (self.car.cl_drs - p_cl)
            cd_eff = p_cd + drs_ef * (self.car.cd_drs - p_cd)
        else:
            drs_ef = ca.DM.zeros(N)
            cl_eff = p_cl
            cd_eff = p_cd

        opti.subject_to(drs >= 0)
        opti.subject_to(drs <= 1)

        # Current speed (k)
        v_k = v[:-1]

        # Race line curvature at current position
        sf = (1 - n[:-1] * kappa_grid[:-1])
        # d(n)/ds = sf * tan(phi)
        dn_ds = sf * ca.tan(phi[:-1])
        # d(phi)/ds = k_car * (ds_car / ds_center) - k_ref
        # ds_car / ds_center = sf / cos(phi)
        dphi_ds = k_car * (sf / ca.cos(phi[:-1])) - kappa_grid[:-1]
        # d(t)/ds = (ds_car / ds_center) / v
        dt_ds = sf / (v_k * ca.cos(phi[:-1])) 

        # Integrate geometry (euler)
        opti.subject_to(n[1:]   == n[:-1]   + dn_ds * ds)
        opti.subject_to(phi[1:] == phi[:-1] + dphi_ds * ds)
        opti.subject_to(T[1:]   == T[:-1]   + dt_ds * ds)

        #limit racing line to 2 m from center
        opti.subject_to(opti.bounded(-1,n,1))
        # limit heading error to ~30 degrees
        opti.subject_to(opti.bounded(-0.5, phi, 0.5)) 
        opti.subject_to(opti.bounded(-0.15, k_car, 0.15)) #limit max curvature

        # =========================
        # Aero
        # =========================
        F_drag = Car.get_aero_drag(self.car, v_k, cd=cd_eff)

        F_max_x, F_max_y = Car.get_friction_limit(
            self.car, v_k,
            mass=p_mass,
            mu_x=p_mu_x,
            mu_y=p_mu_y,
            cl=cl_eff
        )

        #change to car curvature
        # Lateral Force: F_lat = m * v^2 * curvature
        F_lat = p_mass * v_k**2 * ca.fabs(k_car)

        # Longitudinal force
        F_long_k = F_acc + F_brake

        # =========================
        # Pedal constraints
        # =========================
        opti.subject_to(F_acc >= 0)
        opti.subject_to(F_brake <= 0)

        # =========================
        # Friction ellipse
        # =========================
        term_long = (F_long_k / F_max_x)**2
        term_lat  = (F_lat    / F_max_y)**2
        opti.subject_to(term_long + term_lat <= 1)

        # =========================
        # Dynamics
        # =========================
        F_net = F_long_k - F_drag
        acc = F_net / p_mass

        # Speed integration: v^2 = v0^2 + 2*a*ds
        #but we have to use ds car 
        ds_car = ds * sf / ca.cos(phi[:-1])

        v_next_sq = v_k**2 + 2 * acc * ds_car
        opti.subject_to(v[1:]**2 == v_next_sq)

        # Time integration: T = T0 + ds/v
       # opti.subject_to(T[1:] == T[:-1] + ds_car / v_k)

        # Energy integration
        dE_step = (F_acc / self.car.power_eff + F_brake * self.car.regen_eff) * ds_car
        opti.subject_to(E[1:] == E[:-1] + dE_step / 3.6e6)  # J -> kWh

        # =========================
        # Torque limit (30 Nm per wheel)
        # =========================
        opti.subject_to(((self.car.wheel_radius * F_acc)) /
                    (self.car.max_torque * self.car.gear_ratio * 4) < 1)

        # =========================
        # Power limit
        # =========================
        opti.subject_to(((F_acc * v_k) / self.car.power_eff) / self.car.power_max <= 1)

        # Regen limit
        opti.subject_to(((-F_brake * v_k) * self.car.regen_eff) / self.car.brake_max <= 1)

        # Speed limits
        opti.subject_to(v >= 1.0)  # avoid division by zero

        # =========================
        # Initial conditions
        # =========================
        opti.subject_to(v[0] == init_cond.v)
        opti.subject_to(T[0] == init_cond.T)
        opti.subject_to(E[0] == init_cond.E)
        opti.subject_to(n[0] == 0) #start with car on the line
        opti.subject_to(phi[0] == 0) #start with zero heading error
        opti.subject_to(drs[0] == 0)  # start with DRS closed

        # =========================
        # Energy limitation
        # =========================
        if not self.endurance:
            if self.car.energy_limit is not None:
                opti.subject_to(E[-1] <= self.car.energy_limit)
        else:
            target = init_cond.E + init_cond.E_target
            opti.subject_to(E[-1] <= target)

        # =========================
        # Objective function
        # =========================
        J_Time = T[-1]

        F_norm_factor = (self.car.mass * self.car.g)**2
        J_overlap = 0.004 * ca.sum1((F_acc * -F_brake) / F_norm_factor)

        J_reg_steer = 1.0 * ca.sumsqr(k_car[1:] - k_car[:-1])

        W_line = 10
        J_line = W_line * ca.sumsqr(n)

        opti.minimize(J_Time + J_overlap)

        # =========================
        # Initial guess
        # =========================
        if init_guess is not None:
            opti.set_initial(v, 0.9*init_guess['v'])
            opti.set_initial(T, init_guess['t'])
            opti.set_initial(F_acc, 0.9*init_guess['F_acc'])
            opti.set_initial(F_brake,0.9* init_guess['F_brake'])
        else:
            opti.set_initial(v, 15.0)
            opti.set_initial(F_acc, 100.0)

        opti.set_initial(k_car,kappa_grid[:-1])

        # =========================
        # Solver setup
        # =========================
        p_opts = {'expand': True}
        s_opts = {'max_iter': 2000, 'tol': 1e-3, 'print_level': 5}

        opti.solver('ipopt', p_opts, s_opts)

        try:
            sol = opti.solve()

            # =========================
            # Sensitivity analysis
            # =========================
            L = opti.f + ca.dot(opti.lam_g, opti.g)

            sens_m   = sol.value(ca.jacobian(L, p_mass))
            sens_cl  = sol.value(ca.jacobian(L, p_cl))
            sens_cd  = sol.value(ca.jacobian(L, p_cd))
            sens_mux = sol.value(ca.jacobian(L, p_mu_x))
            sens_muy = sol.value(ca.jacobian(L, p_mu_y))

            print("sensitivity results")
            print(f"Mass: {sens_m:+.4f} s/kg")
            print(f"CL:   {sens_cl:+.4f} s/unit")
            print(f"CdA:  {sens_cd:+.4f} s/unit")

            return SolverResults(
                success=True,
                lap_time=sol.value(T[-1]),
                t=sol.value(T),
                v=sol.value(v),
                s=s_sim,
                kappa_car = sol.value(k_car),
                n=sol.value(n),
                xi=sol.value(phi),
                E=sol.value(E),
                f_acc=sol.value(F_acc) / self.car.power_eff,
                f_brake=sol.value(F_brake) * self.car.regen_eff,
                drs_usage=sol.value(drs_ef),
                sens={
                    'dTime_dm': sens_m,
                    'dTime_dCL': sens_cl,
                    'dTime_dCdA': sens_cd,
                    'dTime_dMuX': sens_mux,
                    'dTime_dMuY': sens_muy
                }
            )

        except RuntimeError as e:
            print(f"Solver failed: {e}")
            return SolverResults(False, 0, [], [], [], [], [], [], {})


class StrategyMaker:

    def __init__(self,car,laptime_map,laps,e_target_min,e_target_max):
        self.car = car
        self.laptime_map = laptime_map
        self.laps = laps
        self.e_target_min = e_target_min
        self.e_target_max = e_target_max
        pass
 

    def ef_guess(self,energy_target_endurance):
        #simulate the car with x kWh with max regen to estimate winning EF_score
        N = self.laps
        total_time = 0
        energy_target_lap = energy_target_endurance/self.laps
        SoC_start = 0.8

        #simulate the race
        for i in range(0,N,1):

            SoC_current = SoC_start - i*(energy_target_lap/self.car.acp_capacity)
            laptime = float(self.laptime_map([SoC_current,energy_target_lap,0.94]))
            total_time += laptime
            #print(f"lap {i+1}, laptime {laptime:.2f}, SoC start: {SoC_current:.2f} SoC end: {SoC_start - (i+1)*(energy_target_lap/self.car.acp_capacity):.2f}")
        
        ef_score = total_time**2*energy_target_endurance
        #print(f"Total time: {total_time}")
        #print(f"Ef_score {ef_score}")

        return ef_score

        print(f"Total time: {total_time}")
        #print(f"Ef_score {ef_score}")

    def getPoints(self,regen,energy_target,delta,ef_min,t_min):

        total_time = 0
        SoC_start = 1.0
        energy_target_lap = energy_target/self.laps

         #simulate the race
        for i in range(0,self.laps,1):

            SoC_current = SoC_start - i*(energy_target_lap/self.car.acp_capacity)
            if SoC_current < 0.05: 
                return 0.0
            laptime = float(self.laptime_map([SoC_current,energy_target_lap,regen]))
            total_time += laptime


        #calculate total points from endurance
        T_min_adj = t_min*(1-delta)
        T_max = 1.5 * T_min_adj # Podle tabulky 11 v pravidlech            
        P_max_end = 250.0
        P_min_end = 0.1 * P_max_end

        if total_time > T_max:
            nom_end = 0.0
        elif total_time < T_min_adj:
            nom_end = 1.0
        else:
            nom_end = (T_max - total_time) / (T_max - T_min_adj)
            
        Endurance_points = (P_max_end - P_min_end) * (nom_end**2) + P_min_end

        if total_time > T_max: 
            Endurance_points = 0.0

        P_max_eff = 75.0
        EF_max = 2 * ef_min
        Ef_team = (total_time**2) * energy_target

        if Ef_team > EF_max:
            nom_eff = 0.0
        elif Ef_team < ef_min:
            nom_eff = 1.0
        else:
            nom_eff = (EF_max - Ef_team) / (EF_max - ef_min)
            
        Efficiency_points = P_max_eff * (nom_eff**2)
        

        return Endurance_points + Efficiency_points


    def solve(self,ef_min=0,T_min=0,objective="points",enforce_full_soc = False,):
            opti = ca.Opti()
            N = self.laps

            E_lap = opti.variable(N)

            
            SoC_start = opti.variable()

            if enforce_full_soc:
                opti.subject_to(SoC_start == 1.0)
        
            
            # Stav baterie (SoC) na začátku každého kola + na úplném konci závodu (N+1)
            SoC = opti.variable(N + 1)    

            MIN_SOC = 0.11

            # ==========================================
            # 3. Počáteční podmínky
            # ==========================================
            opti.subject_to(SoC[0] == SoC_start) 
        

            total_time = 0
            
            for k in range(N):
            
                opti.subject_to(SoC[k+1] == SoC[k] - (E_lap[k] / self.car.acp_capacity))
               
                state_vec = ca.vertcat(SoC[k], E_lap[k], self.car.regen_eff)
                laptime = self.laptime_map(state_vec)
    
                total_time += laptime


    
            opti.subject_to(SoC >= MIN_SOC)
            opti.subject_to(SoC_start >= 0.5)
            opti.subject_to(SoC_start <= 1.0)
            #opti.subject_to(SoC <= 1.0)
            
        
            opti.subject_to(E_lap >= self.e_target_min) 
            opti.subject_to(E_lap <= self.e_target_max) 


            # === INITIAL GUESS ===
            E_guess = 0.25
            opti.set_initial(E_lap, E_guess)
            opti.set_initial(SoC_start, 1.0)
            soc_guess = [1.0 - k * (E_guess / self.car.acp_capacity) for k in range(N + 1)]
            opti.set_initial(SoC, soc_guess)
           
        
            # ==========================================
            # Konstanty pro výpočty 
            # ==========================================
            
            T_max = 1.5 * T_min # Podle tabulky 11 v pravidlech
            
            P_max_end = 250.0
            P_min_end = 0.1 * P_max_end 

            # ==========================================
            # Endurance Score
            # ==========================================
            if objective == "points":
                opti.subject_to(total_time <= T_max) 
            
            nom_end = (T_max - total_time) / (T_max - T_min)
            Endurance_points = (P_max_end - P_min_end) * (nom_end**2) + P_min_end

            # ==========================================
            # Efficiency Score
            # ==========================================
            P_max_eff = 75.0
            EF_max = 2 * ef_min
            Ef_team = (total_time**2) * ca.sum1(E_lap)

            if objective == "points":
                opti.subject_to(Ef_team <= EF_max)

            nom_eff = (EF_max - Ef_team) / (EF_max - ef_min)
            Efficiency_points = P_max_eff * (nom_eff**2)
            
            # ==========================================
            # Objective Function
            # ==========================================
            Total_points = Endurance_points + Efficiency_points
            
            if objective == "points":
                opti.minimize(-Total_points)

            elif objective == "time":
                opti.minimize(total_time)

            elif objective == "efficiency":
                opti.minimize(Ef_team)
            
            # ==========================================
            # 8. Solver settings
            # ==========================================
            p_opts = {"expand": True}
            s_opts = {
                "max_iter": 5000, 
                "tol": 1e-3,                   
                "acceptable_tol": 1e-2,           
                "print_level": 0,
                "hessian_approximation": "limited-memory"
            }
            opti.solver('ipopt', p_opts, s_opts)

            try:
                
                sol = opti.solve()
                
                print(f"Celkový čas: {sol.value(total_time):.2f} s")
                
                # Extrakce optimálních hodnot
                opt_soc = sol.value(SoC)
                opt_e_targets = sol.value(E_lap)
                
                # Ošetření pro případ, že N=1 (CasADi vrací skalár místo pole)
                if N == 1:
                    opt_e_targets = [opt_e_targets]
                    opt_soc = [opt_soc, sol.value(SoC)[1]]

                opt_lap_times = []
                
               
                for k in range(N):
                    # Extrakce konkrétních hodnot pro toto kolo
                    soc_k = float(opt_soc[k])
                    e_k = float(opt_e_targets[k])
                    regen_k = float(self.car.regen_eff)
                    
                    # Výpočet času na kolo z mapy
                    lap_time_k = float(sol.value(self.laptime_map(ca.vertcat(soc_k, e_k, regen_k))))
                    opt_lap_times.append(lap_time_k)
                    
                    
                    if lap_time_k < 20.0: 
                        print(f"   Laptime: {lap_time_k:.2f} s")
                        print(f"   Vstupy -> SoC_start: {soc_k:.4f}, E_lap: {e_k:.4f}, Regen: {regen_k:.2f}")
                        return {"sccess":False}

                opt_total_time = sol.value(total_time)

                return {
                    "success": True,
                    "total_time": opt_total_time,
                    "e_targets": opt_e_targets,     
                    "soc_profile": opt_soc,
                    "soc_start": sol.value(SoC_start),
                    "lap_times": opt_lap_times,
                    "points": sol.value(Total_points),
                    "ef_score": sol.value(Ef_team)
                }
                
            except RuntimeError as e:
                print(f"Solver failed: {e}")
                opti.debug.show_infeasibilities()
                return {"success": False}

       
@staticmethod
def calculate_initial_guess(car,track, init_cond: InitConditions, ds=1.0):
        #creates warm start for the solver based on simple forward-backward integration and curve speed limits
       
        total_length = track.length
        N = int(np.round(total_length / ds))
        s_sim = np.linspace(0, total_length, N + 1)
        
        ds_const = total_length / N 
        
        kappa_dm = track.get_curvature(s_sim)
        kappa = np.array(kappa_dm).flatten()
        
        m = car.mass
        g = 9.81
        rho = 1.225
        Cl = car.cl
        Cd = car.cd
        mu_x = car.mu_x - 0.1
        mu_y = car.mu_y - 0.1
        P_max = car.power_max-5000
        F_max_abs = 8000
        
    
        #curve limits vectorized
        denom = m * np.abs(kappa) - mu_y * 0.5 * rho * Cl
        
      
        v_max_car = 40.0 
        v_curve = np.full(N+1, v_max_car)
        
        
        mask = denom > 0
        v_sq_limit = (mu_y * m * g) / denom[mask]
        v_curve[mask] = np.sqrt(v_sq_limit)
        
    
        v_curve = np.minimum(v_curve, v_max_car)

        #forward integration (acceleration)
        v_fwd = np.zeros(N+1)
        v_fwd[0] = max(init_cond.v, 3.0) 
        
        for k in range(N):
            v_curr = v_fwd[k]
            
            # Aero Drag
            F_drag = 0.5 * rho * Cd * v_curr**2
            
            # Grip: F_z = m*g + AeroDown
            F_z = m*g + 0.5 * rho * Cl * v_curr**2
            F_grip = mu_x * F_z
            
            # Power: P = F*v -> F = P/v
            F_power = P_max / max(v_curr, 2.0) * car.power_eff
            
            # Useable accel force
            F_acc_max = min(F_grip, F_power, F_max_abs)
            
            # Net force & accel
            F_net = F_acc_max - F_drag
            acc = F_net / m
            
            # kinematics
            v_next = np.sqrt(max(v_curr**2 + 2 * acc * ds_const, 1.0))
            
            # limit by curve
            v_fwd[k+1] = min(v_next, v_curve[k+1])

        # backward integration (braking)
        v_bwd = np.zeros(N+1)
        v_bwd[-1] = v_curve[-1] 
        
        for k in range(N, 0, -1):
            v_curr = v_bwd[k] 
            
            # Aero
            F_drag = 0.5 * rho * Cd * v_curr**2
            F_z = m*g + 0.5 * rho * Cl * v_curr**2
            
            # Max Brake Force
            F_grip = mu_x * F_z
            F_brake_max = min(F_grip, F_max_abs) 
                    
            acc_back = (F_brake_max + F_drag) / m
            
            v_prev = np.sqrt(v_curr**2 + 2 * acc_back * ds_const)
            
            # Min (Backward, Forward, Curve)
            v_bwd[k-1] = min(v_prev, v_curve[k-1], v_fwd[k-1])

        # Final velocity profile
        v_guess = v_bwd 
        
        #smoothing for IPOPT
        window_size = 5
        v_smooth = np.convolve(v_guess, np.ones(window_size)/window_size, mode='same')
        v_smooth[0] = v_guess[0] 
        v_smooth[-1] = v_guess[-1] 
        
        #calculate forces and energy for the guess
        v_sq_diff = np.diff(v_smooth**2)
        acc_req = v_sq_diff / (2 * ds_const)
        
        v_mid = 0.5 * (v_smooth[:-1] + v_smooth[1:])
        
        # Drag
        F_drag_vec = 0.5 * rho * Cd * v_mid**2
        
        # F
        F_net_req = m * acc_req
        F_long_req = F_net_req + F_drag_vec
        
        # time
        dt_vec = ds_const / np.maximum(v_mid, 1.0)
        T_guess = np.concatenate(([0], np.cumsum(dt_vec))) + init_cond.T
        
        # Inputs splitting
        F_acc_guess = np.maximum(F_long_req, 0)
        F_brake_guess = np.minimum(F_long_req, 0)
        
        # Energy
        P_mech = F_long_req * v_mid
       
        dE = P_mech * dt_vec / 3.6e6 
        E_guess = np.concatenate(([init_cond.E], np.cumsum(dE) + init_cond.E))

        return {
            'v': np.array(v_smooth).flatten(),       
            'T': np.array(T_guess).flatten(),        
            'E': np.array(E_guess).flatten(),        
            'F_acc': np.array(F_acc_guess).flatten(),   
            'F_brake': np.array(F_brake_guess).flatten()
        }

class SimulationWorker(QThread):

    progress = pyqtSignal(int, int)     

    def __init__(self, task: SimulationTask):

            self.error.emit(str(e))

@dataclass
class SolverPayload:

import os
import time
import numpy as np
from scipy.interpolate import griddata
from src.optimization.solver import InitConditions
from .base_task import SimulationTask, SolverPayload

class LaptimeMapTask(SimulationTask):
    def __init__(self, solver, car, track, ds, laps):
        self.solver = solver
        self.car = car
        self.track = track
        self.ds = ds
        self.laps = laps

    def execute(self, progress_callback) -> SolverPayload:
        start = time.time()
        
        acp_targets = np.linspace(3.5, 9.0, num=15, endpoint=True)
        e_targets = np.round(acp_targets / self.laps, 2)
        regen_targets = np.linspace(0.0, 1.1, num=15, endpoint=True)
        soc_targets = np.linspace(0.1, 1.05, num=15, endpoint=True)

        total_steps = len(e_targets) * len(regen_targets) * len(soc_targets)
        results_log = []
        flat_lap_times = []
        current_step = 0
        last_slowest_lap = 40
        prev_res = None

        for soc in soc_targets:
            for e_target in e_targets:
                for regen in regen_targets:
                    current_step += 1
                    progress_callback(current_step, total_steps)
                    
                    if soc * self.car.acp_capacity < e_target:
                        flat_lap_times.append(np.nan)
                        continue
                    
                    self.car.regen_eff = regen
                    self.car.energy_limit = e_target
                    E_start = np.round(self.car.acp_capacity * (1-soc), 2)
                    init_cond = InitConditions(v=self.track.speed[0]-2, T=0.0, E=E_start)
                    
                    res = self.solver.solveLap_rk4(init_cond=init_cond, ds=self.ds, init_guess=prev_res)
                    
                    if res.success:
                        laptime = res.lap_time
                        flat_lap_times.append(laptime)
                        prev_res = res
                        if laptime > last_slowest_lap:
                            last_slowest_lap = laptime

                        res.meta_soc = soc
                        res.meta_energy = e_target
                        res.meta_regen = regen
                        res.meta_mass = self.car.mass
                        results_log.append(res)
                    else:
                        flat_lap_times.append(np.nan) 

        flat_lap_times = np.array(flat_lap_times, dtype=float)
        
        if np.isnan(flat_lap_times).any():
            S, E_grid, R = np.meshgrid(soc_targets, e_targets, regen_targets, indexing='ij')
            points = np.column_stack((S.ravel(), E_grid.ravel(), R.ravel()))
            values = flat_lap_times
            valid_mask = ~np.isnan(values)
            
            if valid_mask.any():
                valid_points = points[valid_mask]
                valid_values = values[valid_mask]
                filled_values = griddata(valid_points, valid_values, points, method='nearest')
                flat_lap_times = filled_values

        end = time.time()
        print(f"Time elapsed: {end - start:.6f} s")

        save_dir = os.path.join("data", "tracks", "lapmaps")
        os.makedirs(save_dir, exist_ok=True)
        file_path = os.path.join(save_dir, f"laptime_map_{self.track.name}.npz")

        np.savez_compressed(
            file_path,
            soc_targets=np.array(soc_targets),
            e_targets=np.array(e_targets),
            regen_targets=np.array(regen_targets),
            flat_lap_times=np.array(flat_lap_times), 
            laps=np.array(self.laps),
            acp_capacity=np.array(self.car.acp_capacity)
        )

        return SolverPayload(
            kind="acp_test",
            data=results_log,
            meta={"laps": self.laps, "car": self.car, "track": self.track}
        )

import numpy as np
from scipy.interpolate import RegularGridInterpolator
from .base_task import SimulationTask, SolverPayload

class StrategyOptTask(SimulationTask):
    def __init__(self, solver, car):
        self.solver = solver
        self.car = car

    def execute(self, progress_callback) -> SolverPayload:
        ef_min_res = self.solver.solve(0,0,objective="efficiency")
        ef_min = ef_min_res["ef_score"] if ef_min_res["success"] else self.solver.ef_guess(4)

        t_min_res = self.solver.solve(0,0,objective="time")
        t_min = t_min_res["total_time"] if t_min_res["success"] else self.solver.best_T_guess(0)

        regen_targets = np.linspace(0.15, 1.0, num=10, endpoint=True)
        delta_targets = np.linspace(0.0, 0.15, num=10, endpoint=True)

        e_opt_grid = np.zeros((len(regen_targets), len(delta_targets)))
        soc_opt_grid = np.zeros((len(regen_targets), len(delta_targets)))

        current_step = 0
        total_steps = len(regen_targets) * len(delta_targets)

        for i, regen_target in enumerate(regen_targets):
            for j, delta_target in enumerate(delta_targets):
                current_step += 1
                progress_callback(current_step, total_steps)
                
                self.car.regen_eff = regen_target
                t_min_adj = t_min * (1 - delta_target)
              
                res = self.solver.solve(ef_min, t_min_adj, objective="points")
                
                if res and res.get("success", False):
                    e_opt_grid[i, j] = np.sum(res["e_targets"])
                    soc_opt_grid[i, j] = res["soc_start"]
                else:
                    e_opt_grid[i, j] = np.nan
                    soc_opt_grid[i, j] = np.nan

        e_interpolant = RegularGridInterpolator((regen_targets, delta_targets), e_opt_grid, method='linear', bounds_error=False, fill_value=np.nan)
        soc_interpolant = RegularGridInterpolator((regen_targets, delta_targets), soc_opt_grid, method='linear', bounds_error=False, fill_value=np.nan)
     
        return SolverPayload(
            kind="endurance_map",
            data=None,
            meta={
                "ef_min": ef_min, "t_min": t_min,
                "e_interpolant": e_interpolant, "soc_interpolant": soc_interpolant,  
                "grid_regen": regen_targets, "grid_delta": delta_targets,
                "z_min_e": np.nanmin(e_opt_grid), "z_max_e": np.nanmax(e_opt_grid),
                "car": self.car, "track": None
            }
        )

class DeploymentOptTask(SimulationTask):
    def __init__(self, solver, car):

        self.car = car

    def execute(self, progress_callback) -> SolverPayload:

    def load_data(self):
        self.runs = []
        for item in self.selected_data:
            try:
                data = np.load(item['path'], allow_pickle=True)
                run_dict = {
                    's': data['s'],
                    'v': data['v'],
                    't': data['t'],
                    'E': data['E'],
                    'x': data['x'], 
                    'y': data['y'], 
                    'f_acc': data['f_acc'],
                    'f_brake': data['f_brake'],
                    'time': float(data['lap_time']),
                    'name': item['legend']
                }
                self.runs.append(run_dict)
                data.close()
            except Exception as e:
                print(f"Error: {e}")

        if not self.runs: return

        # Sort: nejrychlejší první
        self.runs.sort(key=lambda x: x['time'])
        self.ref_run = self.runs[0]

    def load_data(self, filepath):
        try:
            data = np.load(filepath)
            soc_targets = data['soc_targets']
            e_targets = data['e_targets']
            regen_targets = data['regen_targets']
            flat_lap_times = data['flat_lap_times']
            
            shape_3d = (len(soc_targets), len(e_targets), len(regen_targets))
            lap_times_3d = flat_lap_times.reshape(shape_3d)
            
            # Outlier filter
            smoothed_3d = median_filter(lap_times_3d, size=3)
            diff = np.abs(lap_times_3d - smoothed_3d)
            outliers_mask = diff > 5.0
            num_outliers = np.sum(outliers_mask)
            
            if num_outliers > 0:
                print(f"[Data Load] Repaired {num_outliers} outliers via median filter.")
                lap_times_3d[outliers_mask] = smoothed_3d[outliers_mask]
            
            flat_for_casadi = lap_times_3d.ravel(order='F')
            grid_axes = [soc_targets.tolist(), e_targets.tolist(), regen_targets.tolist()]
            interpolant = ca.interpolant('laptime_map', 'bspline', grid_axes, flat_for_casadi.tolist())

            self.min_e_target = min(e_targets)
            self.max_e_target = max(e_targets)
            
            return {
                'interpolant': interpolant,
                'e_fine': np.linspace(min(e_targets), max(e_targets), 50),
                'r_fine': np.linspace(min(regen_targets), 1.0, 50),
                'soc_fine': np.linspace(min(soc_targets), 1.0, 50),
                'r_min': min(regen_targets),
                'r_max': max(regen_targets),
                'z_min': np.nanmin(lap_times_3d),
                'z_max': np.nanmax(lap_times_3d) 
            }
        except Exception as e:
            QMessageBox.critical(self, "Load Error", f"Could not load map from {filepath}:\n{e}")
            return None

    def __init__(self, parent=None):

                font-weight: bold;
            }
            QProgressBar::chunk {
                background-color: #007bff;
                width: 20px; 
            }
        """)
        
        # Přidání do layoutu
        layout.addWidget(self.lbl_gif)   # GIF nahoře
        layout.addWidget(self.lbl_text)  # Text uprostřed
        layout.addWidget(self.progress)  # Bar dole
        
        self.setLayout(layout)
        
        self.current_progress_fmt = None

    def show_loading(self, text="Calculating...", indefinite=True, progress_text_fmt=None):
        """
        Zobrazí overlay.
        Args:
            text: Úvodní text.
            indefinite: 
                True = Single Lap / ACP -> Ukáže GIF, skryje Bar.
                False = Endurance -> Ukáže Bar, skryje GIF.
        """
        if self.parent():
            self.resize(self.parent().size())
            
        self.lbl_text.setText(text)
        
        # Uložení formátu textu
        if progress_text_fmt:
            self.current_progress_fmt = progress_text_fmt
        else:
            self.current_progress_fmt = f"{text} ({{}} / {{}})"

        if indefinite:
            # --- REŽIM GIF (Točící se kolečko / Animace) ---
            self.progress.hide()       # Schováme progress bar
            self.lbl_gif.show()        # Ukážeme GIF
            self.movie.start()         # Spustíme animaci
        else:
            # --- REŽIM PROGRESS BAR (Endurance) ---
            self.lbl_gif.hide()        # Schováme GIF (nebo ho necháme, pokud chceš obojí)
            self.movie.stop()          # Zastavíme animaci (šetří CPU)
            
            self.progress.show()       # Ukážeme progress bar
            self.progress.setRange(0, 100)
            self.progress.setValue(0)

        self.show()
        self.raise_()

    def update_progress(self, current, total):
        # ... (Tato metoda zůstává stejná jako předtím) ...
        if self.current_progress_fmt:
            try:
                self.lbl_text.setText(self.current_progress_fmt.format(current, total))
            except:
                self.lbl_text.setText(f"{current} / {total}")
        
        if self.progress.minimum() == 0 and self.progress.maximum() == 0:
            self.progress.setRange(0, total)
            self.progress.setTextVisible(True)
            self.progress.setFormat("%v / %m") 
            
        self.progress.setValue(current)

    def hide_loading(self):
        self.movie.stop() # Důležité: zastavit GIF při skrytí
        self.hide()

    def plot_track_map(self, track: Track):
        self.mpl_canvas.fig.clf()
        ax = self.mpl_canvas.fig.add_subplot(111)
        ax.set_facecolor('#ffffff') # Bílé pozadí grafu
        x, y = track.x, track.y
        alpha_deg = np.degrees(track.alpha) if hasattr(track, 'alpha') and track.alpha is not None else np.zeros_like(x)

        points = np.array([x, y]).T.reshape(-1, 1, 2)
        segments = np.concatenate([points[:-1], points[1:]], axis=1)

        norm = Normalize(vmin=np.min(alpha_deg), vmax=np.max(alpha_deg))
        lc = LineCollection(segments, cmap='viridis', norm=norm)
        lc.set_array(alpha_deg[:-1])
        lc.set_linewidth(3)
        ax.add_collection(lc)

        ax.set_xlim(x.min() - 20, x.max() + 20)
        ax.set_ylim(y.min() - 20, y.max() + 20)
        ax.set_aspect('equal')
        ax.plot(x[0], y[0], 'o', color='#10b981', markersize=8, markeredgecolor='black') # Moderní zelená tečka
        ax.axis('off')
        
        self.mpl_canvas.fig.tight_layout()
        self.mpl_canvas.draw()

    def on_create_track_clicked(self):
        fname, _ = QFileDialog.getOpenFileName(self, "select telemetry log", "", "MAT Files (*.mat)")
        if not fname: return

        self.loading_overlay.show_loading("parsing log data...", indefinite=True)
        QApplication.processEvents()

        try:
            data = TrackGenerator.filter_data(fname)
        except Exception as e:
            self.loading_overlay.hide_loading()
            QMessageBox.critical(self, "error", f"failed to parse file:\n{e}")
            return
            
        self.loading_overlay.hide_loading()
        if data is None: return

        msg = QMessageBox(self)
        msg.setWindowTitle("select track mode")
        msg.setText("How do you want to define the track?")
        btn_closed = msg.addButton("closed track (endurance)", QMessageBox.ButtonRole.AcceptRole)
        btn_open = msg.addButton("open section (autocross)", QMessageBox.ButtonRole.RejectRole)
        msg.addButton(QMessageBox.StandardButton.Cancel)
        msg.exec()
        
        if msg.clickedButton() == msg.button(QMessageBox.StandardButton.Cancel): return
        is_closed_mode = (msg.clickedButton() == btn_closed)

        lap_info = None
        if is_closed_mode:
            lap_info = TrackGenerator.run_lap_selector(data)
        else:
            lap_info = TrackGenerator.manual_lap_selector(data)

        if not lap_info: return

        default_name = f"track_{lap_info['time']:.2f}s.npz"
        save_path, _ = QFileDialog.getSaveFileName(
            self, "save baked track", 
            os.path.join("data/tracks", default_name), 
            "NumPy Zip (*.npz)"
        )

        if save_path:
            self.loading_overlay.show_loading("smoothing track...", indefinite=True)
            self.track_worker = TrackGenerationWorker(data, lap_info, is_closed_mode, save_path)
            self.track_worker.finished.connect(self.on_track_generation_finished)
            self.track_worker.start()

    def on_track_generation_finished(self, success, message):
        self.loading_overlay.hide_loading()
        if success:
            QMessageBox.information(self, "success", f"track saved successfully!\n{message}")
            self.load_track_list()
            track_name = os.path.splitext(os.path.basename(message))[0]
            idx = self.combo_track.findText(track_name)
            if idx >= 0:
                self.combo_track.setCurrentIndex(idx)
        else:
            QMessageBox.critical(self, "error", f"track generation failed:\n{message}")

    def load_logs(self):
        """Proskenuje složku results a naplní tabulku s checkboxy."""
        self.tree.clear()
        results_dir = "results"
        
        if not os.path.exists(results_dir):
            os.makedirs(results_dir)
            return

        files = glob.glob(os.path.join(results_dir, "*.npz"))
        logs_data = []

        # A) Načtení metadat
        for filepath in files:
            try:
                data = np.load(filepath, allow_pickle=True)
                
                track_name = str(data['track_name']) if 'track_name' in data else "Unknown"
                lap_time = float(data['lap_time']) if 'lap_time' in data else 0.0
                car_name = str(data['car_name']) if 'car_name' in data else "-"
                
                # Zformátujeme datum ze jména souboru (pokud to jde), jinak jen filename
                filename = os.path.basename(filepath)
                
                logs_data.append({
                    'track': track_name,
                    'time': lap_time,
                    'car': car_name,
                    'file': filename,
                    'path': filepath
                })
                data.close()
            except Exception as e:
                print(f"Error reading {filepath}: {e}")
                continue

        # B) Třídění (Track -> Time)
        logs_data.sort(key=lambda x: (x['track'], x['time']))

        # C) Vykreslení
        for item in logs_data:
            row = QTreeWidgetItem([
                item['track'],
                f"{item['time']:.3f}",
                item['car'],
                item['file']
            ])
            
            # --- PŘIDÁNÍ CHECKBOXU ---
            # Nastavíme checkbox do sloupce 0 (Track Name)
            row.setCheckState(0, Qt.CheckState.Unchecked)
            
            # Uložíme si cestu k souboru do skrytých dat
            row.setData(0, Qt.ItemDataRole.UserRole, item['path'])
            
            self.tree.addTopLevelItem(row)

    def on_compare_clicked(self):
        """Zkontroluje výběr a spustí porovnání."""
        selected_data = []
        track_names_set = set()
        
        # 1. Iterace přes všechny řádky
        root = self.tree.invisibleRootItem()
        count = root.childCount()
        
        for i in range(count):
            item = root.child(i)
            
            # Je checkbox zaškrtnutý?
            if item.checkState(0) == Qt.CheckState.Checked:
                path = item.data(0, Qt.ItemDataRole.UserRole)
                track_name = item.text(0)
                lap_time = item.text(1)
                filename = item.text(3)
                
                selected_data.append({
                    'path': path,
                    'track': track_name,
                    'legend': f"{filename} ({lap_time}s)"
                })
                
                track_names_set.add(track_name)

        # 2. VALIDACE: Počet logů
        if len(selected_data) < 2:
            QMessageBox.warning(self, "Not enough data", 
                                "Please select at least 2 logs to compare.")
            return

        # 3. VALIDACE: Shoda tratí
        if len(track_names_set) > 1:
            tracks_list = ", ".join(track_names_set)
            QMessageBox.critical(self, "Track Mismatch", 
                                 f"You cannot compare logs from different tracks!\n\n"
                                 f"Selected tracks: {tracks_list}\n"
                                 f"Please select logs from the same track only.")
            return

        # 4. Success -> Spustit vizualizaci
        # (Zatím jen print, příště uděláme okno)
        print(f"Comparing {len(selected_data)} logs form track '{list(track_names_set)[0]}'...")
        print(f"Comparing {len(selected_data)} logs...")
        
        # Uložíme do self, aby okno nezmizelo
        self.comparison_window = ComparisonPlotWindow(selected_data)
        self.comparison_window.show()

    def calculate_validation_metrics(self, sim_res, ref_data):
        if ref_data is None or ref_data['v'] is None:
            return None

        s_sim = sim_res.s
        s_ref = ref_data['s']
        v_ref_raw = ref_data['v']
        
        ds_ref = np.diff(s_ref, prepend=0)
        v_safe = np.maximum(v_ref_raw, 1.0) 
        dt_ref = ds_ref / v_safe
        t_ref_calculated = np.cumsum(dt_ref)

        v_ref_interp = np.interp(s_sim, s_ref, v_ref_raw)
        t_ref_interp = np.interp(s_sim, s_ref, t_ref_calculated)
        
        p_ref_interp = None
        if ref_data.get('p') is not None:
            p_ref_interp = np.interp(s_sim, s_ref, ref_data['p']) * 1000.0

        delta_v = sim_res.v - v_ref_interp 
        
        t_sim_zeroed = sim_res.t - sim_res.t[0]
        t_ref_zeroed = t_ref_interp - t_ref_interp[0]
        
        delta_t = t_sim_zeroed - t_ref_zeroed
        rmse_v = np.sqrt(np.mean(delta_v**2))
        total_time_diff = t_sim_zeroed[-1] - t_ref_zeroed[-1]
        
        return {
            's': s_sim,
            'v_ref': v_ref_interp,
            't_ref': t_ref_zeroed,
            'p_ref': p_ref_interp,
            'delta_v': delta_v,
            'delta_t': delta_t,
            'rmse_v': rmse_v,
            'diff_total': total_time_diff
        }

    def init_powertrain_tab(self):
        widget = QWidget()

        v_interp = self.res.v[:-1]

        brake_kw = (self.res.f_brake * v_interp) / 1000.0 
        
        plot_p = pg.PlotWidget()
        self._style_plot(plot_p, "Power Usage", "Power", "kW")
        legend_p = plot_p.addLegend(offset=(20, 20), pen=pg.mkPen('#e2e8f0'), brush=pg.mkBrush('#ffffffEE'))

        # --- ENERGY ---
        dt_sim = np.diff(self.res.t)
        pos_power_battery = np.maximum(power_kw, 0) / self.car.power_eff 
        e_step_sim = (pos_power_battery * dt_sim) / 3600.0
        e_cum_sim = np.insert(np.cumsum(e_step_sim), 0, 0.0)

        val_data = None
        e_cum_real = None
        
        if self.track.ref_data is not None:
            val_data = self.calculate_validation_metrics(self.res, self.track.ref_data)
            if val_data and val_data.get('p_ref') is not None:
                p_ref_kw = val_data['p_ref'] / 1000.0 
                plot_p.plot(val_data['s'], p_ref_kw, pen=pg.mkPen(self.C_REF, width=2, style=Qt.PenStyle.DotLine), name="Real Power")
                
                ds = np.diff(val_data['s'])
                v_ref_safe = np.maximum(val_data['v_ref'][:-1], 1.0)
                dt_ref = ds / v_ref_safe
                pos_power_real = np.maximum(p_ref_kw[:-1], 0)
                
                e_step_real = (pos_power_real * dt_ref) / 3600.0
                e_cum_real = np.insert(np.cumsum(e_step_real), 0, 0.0)

        plot_e = pg.PlotWidget()
        self._style_plot(plot_e, "Cumulative Energy (Discharge only)", "Energy Out", "kWh")
        legend_e = plot_e.addLegend(offset=(20, 20), pen=pg.mkPen('#e2e8f0'), brush=pg.mkBrush('#ffffffEE'))
        legend_e.setLabelTextColor('#0f172a')
        
        e_curve = plot_e.plot(self.res.s, e_cum_sim, pen=pg.mkPen(self.C_ENERGY, width=3), name="Sim")
        
        # Výplň energie
        dummy_zero = pg.PlotCurveItem(self.res.s, np.zeros_like(self.res.s), pen=None)
        fill_color = pg.mkColor(self.C_ENERGY)
        fill_color.setAlpha(30)
        plot_e.addItem(pg.FillBetweenItem(e_curve, dummy_zero, brush=fill_color))
        
        if e_cum_real is not None:
            plot_e.plot(val_data['s'], e_cum_real, pen=pg.mkPen(self.C_REF, width=2, style=Qt.PenStyle.DashLine), name="Real discharge")

        layout.addWidget(plot_p)
        layout.addWidget(plot_e)

            dists = np.sqrt((self.map_data['x'] - mx)**2 + (self.map_data['y'] - my)**2)
            idx = np.argmin(dists) 
            
            if dists[idx] < 20.0:

                    f"<b>Power:</b> <span style='color: {'#10b981' if p_val > 0 else '#ef4444'};'>{p_val:.1f} kW</span>"
                    f"</div>"
                )
                self.cursor_label.setHtml(html_text)
                self.cursor_label.setPos(self.map_data['x'][idx], self.map_data['y'][idx])
                self.cursor_label.setVisible(True)
                self.v_line.setPos(self.map_data['x'][idx])
                self.h_line.setPos(self.map_data['y'][idx])

                self.h_line.setVisible(True)

                self.cursor_label.setVisible(False)
                self.v_line.setVisible(False)
                self.h_line.setVisible(False)

    def reset_scenario(self):
        for spin in self.inputs.values():
            spin.blockSignals(True) 
            spin.setValue(0.0)
            spin.blockSignals(False)
        self.update_scenario()

    def update_scenario(self):
        total_dt = 0.0
        
        for name, spin in self.inputs.items():
            pct = spin.value()
            data = self.scenario_data[name]
            d_param = data['val'] * (pct / 100.0)
            d_time = data['sens'] * d_param
            total_dt += d_time
            
            color = "#10b981" if d_time < 0 else "#ef4444" if d_time > 0 else "#94a3b8"
            sign = "+" if d_time > 0 else ""
            self.deltas_labels[name].setText(f"<span style='color:{color}; background-color: transparent;'>{sign}{d_time:.4f} s</span>")

        base_time = self.res.lap_time
        new_time = base_time + total_dt
        
        diff_color = "#10b981" if total_dt < 0 else "#ef4444" if total_dt > 0 else "#475569"
        diff_sign = "+" if total_dt > 0 else ""

        html = f"""
        <div style='text-align: center;'>
            <div style='color: #64748b; font-size: 14px; text-transform: uppercase; font-weight: bold;'>Predicted Lap Time</div>
            <div style='font-size: 56px; font-weight: bold; color: {diff_color}; margin: 15px 0;'>
                {new_time:.3f} s
            </div>
            <div style='font-size: 24px; color: {diff_color}; font-weight: bold;'>
                ({diff_sign}{total_dt:.3f} s)
            </div>
            <div style='margin-top: 25px; font-size: 15px; color: #64748b; border-top: 1px solid #cbd5e1; padding-top: 15px;'>
                Baseline: {base_time:.3f} s
            </div>
        </div>
        """
        self.lbl_scenario_results.setText(html)

class Universal3DWindow(QWidget):
    def __init__(self, x_grid, y_grid, z_grid, x_label, y_label, z_label, title, parent=None):
        super().__init__(parent)
       
        self.x_label = x_label
        self.y_label = y_label
        self.z_label = z_label
        self.plot_title = title
        
        # Layout
        layout = QVBoxLayout()
        self.setLayout(layout)
        
        # Matplotlib figure
        self.fig = Figure(figsize=(10, 8), dpi=100)
        self.canvas = FigureCanvasQTAgg(self.fig)
        self.toolbar = NavigationToolbar2QT(self.canvas, self)
        
        layout.addWidget(self.toolbar)
        layout.addWidget(self.canvas)
        

        self.plot_3d(x_grid, y_grid, z_grid)

    def plot_3d(self, X_raw, Y_raw, Z_raw):
        self.fig.clear()
        
        ax3d = self.fig.add_subplot(1, 1, 1, projection='3d')
        
        x_flat = X_raw.flatten()
        y_flat = Y_raw.flatten()
        z_flat = Z_raw.flatten()
        
        mask = ~np.isnan(z_flat)
        x_valid = x_flat[mask]
        y_valid = y_flat[mask]
        z_valid = z_flat[mask]
        
        if len(z_valid) < 4:
            print("Not enough points for interpolation")
            return

        resolution = 100 
        xi = np.linspace(min(x_valid), max(x_valid), resolution)
        yi = np.linspace(min(y_valid), max(y_valid), resolution)
        XI, YI = np.meshgrid(xi, yi)
        
        # Cubic interpolation
        ZI = griddata((x_valid, y_valid), z_valid, (XI, YI), method='cubic')
        
        # Plot 3D Surface
        surf = ax3d.plot_surface(XI, YI, ZI, 
                               cmap='viridis', edgecolor='none', 
                               linewidth=0, antialiased=True, alpha=0.9,
                               rcount=resolution, ccount=resolution)
        
      
        ax3d.set_xlabel(self.x_label)
        ax3d.set_ylabel(self.y_label)
        ax3d.set_zlabel(self.z_label)
        ax3d.set_title(self.plot_title)
        
        z_min = np.nanmin(ZI)
        ax3d.contourf(XI, YI, ZI, zdir='z', offset=z_min, cmap='viridis', alpha=0.3)
        ax3d.set_zlim(z_min, np.nanmax(ZI))

        self.fig.colorbar(surf, ax=ax3d, shrink=0.5, aspect=10, label=self.z_label)
        ax3d.view_init(elev=25, azim=-135)

        self.fig.tight_layout()
        self.canvas.draw()

    def get_laptime_functions(self):
        x_vals = sorted(list(set([r.meta_energy*self.laps for r in self.results]))) 
        y_vals = sorted(list(set([r.meta_regen for r in self.results])))  
        
        nx = len(x_vals)
        ny = len(y_vals)
        
        Z_time = np.zeros((nx, ny))
        Z_eff = np.zeros((nx, ny))
        Z_time[:] = np.nan
        Z_eff[:] = np.nan
        
        x_map = {val: i for i, val in enumerate(x_vals)}
        y_map = {val: i for i, val in enumerate(y_vals)}
        
        for r in self.results:
            ix = x_map.get(r.meta_energy*self.laps)
            iy = y_map.get(r.meta_regen)
            if ix is not None and iy is not None:
                t = r.lap_time
                Z_time[ix, iy] = t
                Z_eff[ix, iy] = (t**2) * (r.meta_energy*self.laps)
        
        if np.isnan(Z_time).any():
            Z_time = np.nan_to_num(Z_time, nan=np.nanmax(Z_time))
            Z_eff = np.nan_to_num(Z_eff, nan=np.nanmax(Z_eff))

        interp_method = 'cubic' if (nx >= 4 and ny >= 4) else 'linear'

        interp_time_obj = RegularGridInterpolator((x_vals, y_vals), Z_time, method=interp_method, bounds_error=False, fill_value=None)
        
        def laptime_func(E_target, Regen):
            E_arr = np.atleast_1d(E_target)
            R_arr = np.atleast_1d(Regen)
            if E_arr.shape != R_arr.shape:
                E_arr, R_arr = np.broadcast_arrays(E_arr, R_arr)
                
            pts = np.column_stack((E_arr.ravel(), R_arr.ravel()))
            res = interp_time_obj(pts).reshape(E_arr.shape)
            return float(res) if res.size == 1 else res

        return laptime_func, None

    def create_acp_plot(self):

        ny = len(y_vals)
        X, Y = np.meshgrid(x_vals, y_vals)

        Z[:] = np.nan
        
        x_map = {val: i for i, val in enumerate(x_vals)}
        y_map = {val: i for i, val in enumerate(y_vals)}
        
        for r in self.results:
            ix = x_map.get(r.meta_energy*self.laps)
            iy = y_map.get(r.meta_regen)
            if ix is not None and iy is not None:
                Z[iy, ix] = r.lap_time

        return Universal3DWindow(X, Y, Z, 
                                 x_label='Total energy target [kWh]', 
                                 y_label='Regen percentage [%]', 
                                 z_label='Lap Time [s]', 
                                 title='Lap time sensitivity')

    def create_laptimes_energy(self):
        laptime_func, _ = self.get_laptime_functions()
        
        all_times = np.array([r.lap_time for r in self.results])
        all_energies = np.array([r.meta_energy*self.laps for r in self.results])
        all_eff_factors = (all_times**2) * all_energies
        
        T_min_global = np.min(all_times) * (1 - 0.07)
        Best_Eff_global = np.min(all_eff_factors)
        
        lt_widget = pg.GraphicsLayoutWidget(title="Energy vs laptime and points")
        lt_widget.setBackground('w')
        
        p1 = lt_widget.addPlot(title="Lap Time [s]")
        p1.setLabel('left', "Time", units='s')
        p1.addLegend(offset=(30, 30))

        lt_widget.nextRow()

        p2 = lt_widget.addPlot(title="Total points vs energy used")
        p2.setLabel('left', "Points", units='pts')
        p2.setLabel('bottom', "Total Energy target for endurance", units='kWh')
        p2.setXLink(p1)
        p2.addLegend(offset=(30, 30))
        p2.plot([0], [0], pen=pg.mkPen('k', width=2, style=Qt.PenStyle.SolidLine), name="Total Score")

        min_e, max_e = np.min(all_energies), np.max(all_energies)
        regens_set = sorted(list(set([r.meta_regen for r in self.results])))
        min_r, max_r = min(regens_set), max(regens_set)
        
        x_smooth = np.linspace(min_e, max_e, 200)
        regen_levels = np.linspace(min_r, max_r, 4) 
        
        global_max_points = -1
        global_opt_energy = 0
        
        COLORS = [QColor(0, 114, 178), QColor(213, 94, 0), QColor(0, 158, 115), QColor(204, 121, 167)]
        
        for i, r_val in enumerate(regen_levels):
            y_time = laptime_func(x_smooth, np.full_like(x_smooth, r_val))
            y_eff_factor = (y_time**2) * x_smooth
            
            pts_time = np.zeros_like(y_time)
            T_max = T_min_global * 1.5
            mask_t = y_time <= T_max
            pts_time[mask_t] = (250 - 25) * ((T_max - y_time[mask_t]) / (T_max - T_min_global))**2 + 25
            
            pts_eff = np.zeros_like(y_eff_factor)
            Ef_max = (Best_Eff_global * 2) * 1.2
            mask_e = y_eff_factor <= Ef_max
            pts_eff[mask_e] = 75 * ((Ef_max - y_eff_factor[mask_e]) / (Ef_max - Best_Eff_global))**2
            
            pts_total = pts_time + pts_eff
            
            local_idx = np.argmax(pts_total)
            local_opt_energy = x_smooth[local_idx]
            local_max_points = pts_total[local_idx]
            
            if local_max_points > global_max_points:
                global_max_points = local_max_points
                global_opt_energy = local_opt_energy
            
            base_color = COLORS[i % len(COLORS)]
            label_text = f"Regen {r_val*100:.0f}%"
            
            p1.plot(x_smooth, y_time, pen=pg.mkPen(color=base_color, width=3), name=label_text)
            p2.plot(x_smooth, pts_total, pen=pg.mkPen(color=base_color, width=3, style=Qt.PenStyle.SolidLine))
            
            p2.plot([local_opt_energy], [local_max_points], symbol='o', symbolSize=10, 
                    symbolBrush=base_color, symbolPen='k', name=None)

        v_line = pg.InfiniteLine(pos=global_opt_energy, angle=90, pen=pg.mkPen('b', width=2, style=Qt.PenStyle.DashDotLine))
        p2.addItem(v_line)
        p1.addItem(pg.InfiniteLine(pos=global_opt_energy, angle=90, pen=pg.mkPen('b', width=2, style=Qt.PenStyle.DashDotLine)))

        info_html = (f"<div style='color:black; background-color: rgba(220,220,255,0.85); "
                     f"padding:5px; border:2px solid blue; font-weight:bold;'>"
                     f"Global Max Points: {global_max_points:.1f} pts<br>"
                     f"Opt Total Energy: {global_opt_energy:.2f} kWh </div>")
        
        text = pg.TextItem(html=info_html, anchor=(0.5, 0))
        text.setPos(global_opt_energy, global_max_points)
        p2.addItem(text)
        
        return lt_widget

    def create_soc_vs_laptime_plot(self):
        # 1. Zjistíme unikátní hodnoty z celého datasetu
        e_targets = sorted(list(set([r.meta_energy for r in self.all_results])))
        regens = sorted(list(set([r.meta_regen for r in self.all_results])))
        socs = sorted(list(set([r.meta_soc for r in self.all_results])))
        
        if not e_targets or not regens or not socs:
            return QWidget()
            
        # 2. Vybereme fixní hodnoty (prostřední E_target a maximální Regen)
        fixed_e_target = e_targets[len(e_targets) // 2]
        fixed_regen = regens[-1]
        
        # 3. Vyfiltrujeme data
        plot_data = []
        for r in self.all_results:
            # Porovnáváme floaty s tolerancí (abs < 1e-5)
            if abs(r.meta_energy - fixed_e_target) < 1e-5 and abs(r.meta_regen - fixed_regen) < 1e-5:
                plot_data.append((r.meta_soc, r.lap_time))
                
        # Seřadíme podle SoC (od nejmenšího po největší)
        plot_data.sort(key=lambda x: x[0])
        
        x_vals = [d[0] * 100 for d in plot_data] # Převod SoC na procenta (Osa X)
        y_vals = [d[1] for d in plot_data]       # Laptime (Osa Y)
        
        # 4. Vytvoření PyQtGraph widgetu
        title = f"SoC vs Laptime (Fixed Target = {fixed_e_target*self.laps:.2f} kWh, Regen = {fixed_regen*100:.0f}%)"
        plot_widget = pg.PlotWidget(title=title)
        plot_widget.setBackground('w')
        
        plot_widget.setLabel('bottom', "State of Charge (SoC)", units="%")
        plot_widget.setLabel('left', "Lap Time", units="s")

                         symbolBrush=QColor(0, 114, 178), symbolPen='k') # Modré body s černým okrajem
        
        # 6. Invertování osy X (aby 100% bylo vlevo a 0% vpravo, jak se baterie vybíjí)
        plot_widget.getViewBox().invertX(True)
        
        return plot_widget

import json
import os
import glob

class ConfigManager:
    # define the directory for storing vehicle profiles
    VEHICLE_DIR = os.path.join("src", "core", "vehicles")

    @staticmethod
    def _ensure_dir():
        """creates the directory if it doesn't exist."""
        if not os.path.exists(ConfigManager.VEHICLE_DIR):
            os.makedirs(ConfigManager.VEHICLE_DIR)

    @staticmethod
    def get_available_vehicles() -> list:
        """returns a list of saved vehicle names (without .json extension)."""
        ConfigManager._ensure_dir()
        files = glob.glob(os.path.join(ConfigManager.VEHICLE_DIR, "*.json"))
        names = [os.path.splitext(os.path.basename(f))[0] for f in files]
        return sorted(names)

    @staticmethod
    def load_vehicle(name: str, default_config: dict) -> dict:
        """loads a specific vehicle's config and merges it with defaults."""
        ConfigManager._ensure_dir()
        filepath = os.path.join(ConfigManager.VEHICLE_DIR, f"{name}.json")
        
        if not os.path.exists(filepath):
            return default_config.copy()
        
        try:
            with open(filepath, 'r', encoding='utf-8') as f:
                saved_config = json.load(f)
            
            merged_config = default_config.copy()
            merged_config.update(saved_config)
            return merged_config
            
        except Exception as e:
            print(f"failed to load vehicle '{name}': {e}. using defaults.")
            return default_config.copy()

    @staticmethod
    def save_vehicle(name: str, config_dict: dict):
        """saves the current config dictionary to a specific vehicle's json file."""
        ConfigManager._ensure_dir()
        filepath = os.path.join(ConfigManager.VEHICLE_DIR, f"{name}.json")
        try:
            with open(filepath, 'w', encoding='utf-8') as f:
                json.dump(config_dict, f, indent=4)
        except Exception as e:
            print(f"failed to save vehicle '{name}': {e}")

import numpy as np
import matplotlib.pyplot as plt

# ---------------------------
# Simulation parameters
# ---------------------------
N_SIM = 50     # number of Monte Carlo runs
N_LAPS = 22     # laps per run
REGEN_INIT = 0.94

BASE_DRIFT = -0.0015
SIGMA = 0.025


# ---------------------------
# Regen update function
# ---------------------------
def update_regen_usage(regen_eff, base_drift=-0.001, sigma=0.01):
    
    
    noise = np.random.normal(0.0, sigma)
    regen_eff += base_drift + noise
    regen_eff = np.clip(regen_eff, 0.5, 1.0)

    return regen_eff

# ---------------------------
# Run Monte Carlo simulations
# ---------------------------
all_runs = np.zeros((N_SIM, N_LAPS))

for sim in range(N_SIM):
    regen_eff = REGEN_INIT

    for lap in range(N_LAPS):

        

        regen_eff = update_regen_usage(
            regen_eff,
            BASE_DRIFT,
            SIGMA
        )

        all_runs[sim, lap] = regen_eff

# ---------------------------
# Compute statistics
# ---------------------------
mean_curve = np.mean(all_runs, axis=0)
std_curve = np.std(all_runs, axis=0)

# ---------------------------
# Plot
# ---------------------------
plt.figure(figsize=(10, 6))

# individual runs
for i in range(N_SIM):
    plt.plot(all_runs[i], alpha=0.3)

# mean
plt.plot(mean_curve, linewidth=3, label="Mean regen_eff")


plt.title("Monte Carlo: Regen Efficiency Drift Over Laps")
plt.xlabel("Lap")
plt.ylabel("regen_eff")
plt.ylim(0.45, 1.02)
plt.grid(True)
plt.legend()

plt.show()

        lap_time=res.lap_time,

import numpy as np
import matplotlib.pyplot as plt

# ---------------------------
# CONFIG
# ---------------------------
Total_Laps = 22
Num_Simulations = 50

Base_Mu_X = 1.3  # target optimal grip

Tire_Model = {
    "Cold_Factor": 0.6,   # start grip fraction
    "Warmup_Tau": 1.5,    # warmup time constant
    "Noise_Std": 0.016     # driver inconsistency
}

# ---------------------------
# PREP
# ---------------------------
laps_vector = np.arange(1, Total_Laps + 1)

mu_optimal = Base_Mu_X
mu_cold = Base_Mu_X * Tire_Model["Cold_Factor"]

plt.figure(figsize=(10, 6))
plt.grid(True)

# ---------------------------
# MONTE CARLO SIM
# ---------------------------
for sim in range(Num_Simulations):
    grip_history = np.zeros(Total_Laps)

    for lap in range(Total_Laps):
        # Exponential warmup curve
        warmup_progress = 1 - np.exp(-(lap + 1) / Tire_Model["Warmup_Tau"])

        current_base_mu = mu_cold + (mu_optimal - mu_cold) * warmup_progress

        # Noise
        noise = Tire_Model["Noise_Std"] * np.random.randn()

        mu_lap = current_base_mu + noise

        # Clip grip lower bound
        min_mu = mu_cold * 0.9
        mu_lap = max(mu_lap, min_mu)

        grip_history[lap] = mu_lap

    # Plot each simulation
    plt.plot(laps_vector, grip_history, color=(0.2, 0.6, 1.0, 0.3), linewidth=1.5)

# ---------------------------
# IDEAL CURVE (NO NOISE)
# ---------------------------
ideal_curve = mu_cold + (mu_optimal - mu_cold) * (1 - np.exp(-laps_vector / Tire_Model["Warmup_Tau"]))
plt.plot(laps_vector, ideal_curve, "r--", linewidth=2, label="Ideal warmup")

plt.axhline(Base_Mu_X, color="black", linestyle="-", label="Optimal Grip")

# ---------------------------
# PLOT STYLING
# ---------------------------
plt.title(
    f"Tire evolution model (Start: {Tire_Model['Cold_Factor']*100:.0f}%, Tau: {Tire_Model['Warmup_Tau']:.1f})"
)
plt.xlabel("Lap")
plt.ylabel("Longitudinal Friction (μx)")
#plt.ylim([Base_Mu_X * 0.8, Base_Mu_X * 1.05])
plt.legend()

plt.show()
